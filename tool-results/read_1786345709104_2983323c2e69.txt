    97→    97→function HorizontalBookRow({
    98→    98→  books,
    99→    99→  title,

   103→   103→}: {
   104→   104→  books: Book[];

   106→   106→  subtitle: string;
   107→   107→  viewAllLabel: string;

   109→   109→}) {
   110→   110→  const scrollRef = useRef<HTMLDivElement>(null);
   111→   111→
   112→   112→  const scroll = (dir: 'left' | 'right') => {
   113→   113→    if (!scrollRef.current) return;
   114→   114→    const amount = 340;
   115→   115→    scrollRef.current.scrollBy({
   116→   116→      left: dir === 'left' ? -amount : amount,
   117→   117→      behavior: 'smooth',
   118→   118→    });
   119→   119→  };
   120→   120→
   121→   121→  if (books.length === 0) return null;
   122→   122→
   123→   123→  return (
   124→   124→    <section className="py-16 sm:py-20">
   125→   125→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   126→   126→        <SectionHeader
   127→   127→          title={title}
   128→   128→          subtitle={subtitle}
   129→   129→          action={{ label: viewAllLabel, onClick: onViewAll }}
   130→   130→        />
   131→   131→      </div>
   132→   132→
   133→   133→      <div className="relative group/row">
   134→   134→        {/* Left arrow */}
   135→   135→        <button
   136→   136→          onClick={() => scroll('left')}
   137→   137→          className="absolute left-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   138→   138→          aria-label="Scroll left"
   139→   139→        >
   140→   140→          <ChevronLeft className="h-5 w-5" />
   141→   141→        </button>
   142→   142→        {/* Right arrow */}
   143→   143→        <button
   144→   144→          onClick={() => scroll('right')}
   145→   145→          className="absolute right-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   146→   146→          aria-label="Scroll right"
   147→   147→        >
   148→   148→          <ChevronRight className="h-5 w-5" />
   149→   149→        </button>
   150→   150→
   151→   151→        <div
   152→   152→          ref={scrollRef}
   153→   153→          className="flex gap-5 overflow-x-auto scrollbar-hide px-4 sm:px-6 lg:px-8 pb-4 snap-x snap-mandatory"
   154→   154→        >
   155→   155→          {books.map((book, i) => (
   156→   156→            <div key={book.id} className="w-[180px] sm:w-[200px] shrink-0 snap-start">
   157→   157→              <BookCard book={book} index={i} />
   158→   158→            </div>
   159→   159→          ))}
   160→   160→        </div>
   161→   161→      </div>
   162→   162→    </section>
   163→   163→  );
   164→   164→}
