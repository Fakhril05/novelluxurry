    58→

    60→  const { items, getTotalPrice, getDiscountSavings, clearCart } = useCartStore();

    62→  const lang = (locale ?? 'id') as Locale;
    63→
    64→  // Step state
    65→  const [currentStep, setCurrentStep] = useState(0);
    66→

    68→  const [form, setForm] = useState({
    69→    name: '',
    70→    phone: '',
    71→    address: '',
    72→    city: '',
    73→    postalCode: '',
    74→  });

    76→  const [payment, setPayment] = useState('bank-transfer');

    78→  const [voucherLoading, setVoucherLoading] = useState(false);
    79→  const [voucherDiscount, setVoucherDiscount] = useState(0);
    80→  const [voucherApplied, setVoucherApplied] = useState(false);
    81→  const [voucherError, setVoucherError] = useState('');
    82→  const [submitting, setSubmitting] = useState(false);
    83→  const [formErrors, setFormErrors] = useState<Record<string, string>>({});
    84→

    86→  const shippingRef = useRef<HTMLDivElement>(null);
    87→  const confirmRef = useRef<HTMLDivElement>(null);
    88→
    89→  const subtotal = getTotalPrice();

    91→  const shippingCost = EXPEDITIONS.find((e) => e.id === expedition)?.price ?? 0;
    92→  const total = Math.max(0, subtotal - voucherDiscount + shippingCost);
    93→

    95→  useEffect(() => {
    96→    if (isAuthenticated && user) {
    97→      setForm((prev) => ({
    98→        ...prev,
    99→        name: user.name || '',
   100→        phone: user.phone || '',
   101→        address: user.address || '',
   102→        city: user.city || '',
   103→        postalCode: user.postalCode || '',

   113→      const res = await fetch('/api/vouchers/validate', {
   114→        method: 'POST',

   116→        body: JSON.stringify({ code: voucherCode.trim().toUpperCase(), total: subtotal }),
   117→      });

   120→        setVoucherDiscount(data.discount);
   121→        setVoucherApplied(true);

   130→        setVoucherApplied(false);

   145→    if (!form.city.trim()) errors.city = lang === 'id' ? 'Kota wajib diisi' : 'City is required';
   146→    if (!form.postalCode.trim()) errors.postalCode = lang === 'id' ? 'Kode pos wajib diisi' : 'Postal code is required';
   147→    setFormErrors(errors);
   148→    return Object.keys(errors).length === 0;
   149→  };

   154→        toast.error(lang === 'id' ? 'Lengkapi data pengiriman' : 'Please complete shipping information');

   167→      toast.error(lang === 'id' ? 'Keranjang kosong' : 'Cart is empty');
   168→      return;
   169→    }
   170→    if (!validateForm()) {
   171→      toast.error(lang === 'id' ? 'Lengkapi data pengiriman' : 'Please complete shipping information');
   172→      return;

   178→        items: items.map((item) => ({
   179→          bookId: item.book.id,
   180→          title: item.book.title,
   181→          quantity: item.quantity,
   182→          price: item.book.discountPrice ?? item.book.price,
   183→          format: item.book.format,
   184→        })),
   185→        shippingName: form.name,
   186→        shippingPhone: form.phone,
   187→        shippingAddr: form.address,
   188→        shippingCity: form.city,
   189→        shippingCode: form.postalCode,

   192→        paymentMethod: payment,

   194→        voucherCode: voucherApplied ? voucherCode.toUpperCase() : null,
   195→        voucherDisc: voucherDiscount,

   197→        discount: savings + voucherDiscount,
   198→        total,

   201→      const res = await fetch('/api/orders', {
   202→        method: 'POST',
   203→        headers: { 'Content-Type': 'application/json' },
   204→        body: JSON.stringify(orderData),
   205→      });
   206→
   207→      const data = await res.json();
   208→
   209→      if (data.success) {
   210→        toast.success(
   211→          lang === 'id' ? 'Pesanan berhasil dibuat!' : 'Order placed successfully!',

   215→        setPage('dashboard', { tab: 'orders' });
   216→      } else {
   217→        toast.error(data.message || (lang === 'id' ? 'Gagal membuat pesanan' : 'Failed to place order'));
   218→      }
   219→    } catch {
   220→      toast.error(lang === 'id' ? 'Terjadi kesalahan' : 'An error occurred');
   221→    } finally {
   222→      setSubmitting(false);
   223→    }
   224→  };
   225→
   226→  if (items.length === 0) {
   227→    return (
   228→      <div className="flex min-h-[60vh] flex-col items-center justify-center gap-4 px-4">
   229→        <motion.div
   230→          initial={{ scale: 0.8, opacity: 0 }}
   231→          animate={{ scale: 1, opacity: 1 }}
   232→          className="flex h-24 w-24 items-center justify-center rounded-full bg-muted"
   233→        >
   234→          <Package className="h-12 w-12 text-muted-foreground/50" />
   235→        </motion.div>
   236→        <p className="text-muted-foreground font-medium">{t('cart.empty', lang)}</p>
   237→        <Button
   238→          variant="outline"
   239→          onClick={() => setPage('catalog')}
   240→          className="border-[#D4AF37] text-[#D4AF37] hover:bg-[#D4AF37]/10"

   242→          {t('cart.continue', lang)}

   244→      </div>
   245→    );
   246→  }
   247→
   248→  const updateField = (field: string, value: string) => {

   250→    if (formErrors[field]) {
   251→      setFormErrors((prev) => {
   252→        const next = { ...prev };
   253→        delete next[field];
   254→        return next;
   255→      });
   256→    }
   257→  };
   258→
   259→  const selectedExpedition = EXPEDITIONS.find((e) => e.id === expedition);
   260→
   261→  return (
   262→    <div className="mx-auto max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   263→      {/* Back Button + Title */}
   264→      <motion.div
   265→        initial={{ opacity: 0, y: -10 }}
   266→        animate={{ opacity: 1, y: 0 }}
   267→        className="mb-6"
   268→      >
   269→        <button
   270→          onClick={() => setPage('catalog')}
   271→          className="mb-4 flex items-center gap-1.5 text-sm text-muted-foreground transition-colors hover:text-foreground"
   272→        >
   273→          <ArrowLeft className="h-4 w-4" />
   274→          {t('general.back', lang)}
   275→        </button>
   276→        <h1 className="font-heading text-2xl font-bold sm:text-3xl">{t('checkout.title', lang)}</h1>
   277→      </motion.div>
   278→
   279→      {/* ─── Progress Stepper ─── */}

   290→          <div
   291→            className="absolute top-5 left-0 h-0.5 transition-all duration-500 ease-out"

   294→              backgroundColor: GOLD,
   295→            }}
   296→          />
   297→
   298→          <div className="relative flex items-center justify-between">
   299→            {STEPS.map((step, idx) => {

   302→              const isActive = idx === currentStep;

   305→                  key={step.key}
   306→                  onClick={() => {

   308→                  }}
   309→                  className="group flex flex-col items-center gap-2"
   310→                  type="button"

   312→                  <div
   313→                    className={`relative z-10 flex h-10 w-10 items-center justify-center rounded-full border-2 transition-all duration-300 sm:h-11 sm:w-11 ${
   314→                      isCompleted
   315→                        ? `border-[${GOLD}] bg-[${GOLD}] text-white shadow-lg shadow-[${GOLD}]/20`
   316→                        : isActive
   317→                          ? `border-[${GOLD}] bg-[${GOLD}]/10 text-[${GOLD}] ring-4 ring-[${GOLD}]/10`
   318→                          : 'border-muted-foreground/25 bg-background text-muted-foreground/50'
   319→                    }`}
   320→                    style={
   321→                      isCompleted
   322→                        ? { borderColor: GOLD, backgroundColor: GOLD, color: '#fff' }
   323→                        : isActive
   324→                          ? { borderColor: GOLD, color: GOLD }
   325→                          : undefined
   326→                    }
   327→                  >
   328→                    {isCompleted ? (
   329→                      <Check className="h-4 w-4 sm:h-5 sm:w-5" strokeWidth={3} />
   330→                    ) : (
   331→                      <Icon className="h-4 w-4 sm:h-5 sm:w-5" />
   332→                    )}
   333→                  </div>
   334→                  <span
   335→                    className={`text-xs font-medium transition-colors sm:text-sm ${
   336→                      isCompleted || isActive
   337→                        ? 'text-foreground'
   338→                        : 'text-muted-foreground/50'
   339→                    }`}
   340→                  >
   341→                    {t(step.labelId, lang)}
   342→                  </span>
   343→                </button>
   344→              );
   345→            })}
   346→          </div>
   347→        </div>
   348→      </motion.div>
   349→
   350→      {/* ─── Two-Column Layout ─── */}
   351→      <div className="grid gap-8 lg:grid-cols-3">
   352→        {/* Left Column - Forms */}

   358→                key="step-address"

   364→              >
   365→                <Card className="overflow-hidden">
   366→                  <CardContent className="p-5 sm:p-6">
   367→                    <div className="mb-5 flex items-center gap-2.5">
   368→                      <div
   369→                        className="flex h-8 w-8 items-center justify-center rounded-full"
   370→                        style={{ backgroundColor: `${GOLD}15` }}
   371→                      >
   372→                        <MapPin className="h-4 w-4" style={{ color: GOLD }} />
   373→                      </div>
   374→                      <h2 className="font-heading text-lg font-semibold">{t('checkout.address', lang)}</h2>
   375→                    </div>

   379→                          {t('checkout.name', lang)} <span className="text-red-500">*</span>

   381→                        <Input
   382→                          id="name"
   383→                          value={form.name}

   385→                          placeholder={lang === 'id' ? 'Masukkan nama lengkap' : 'Enter full name'}

   389→                          <p className="text-xs text-red-500">{formErrors.name}</p>
   390→                        )}
   391→                      </div>
   392→                      <div className="space-y-1.5">
   393→                        <Label htmlFor="phone" className="text-sm font-medium">
   394→                          {t('checkout.phone', lang)} <span className="text-red-500">*</span>
   395→                        </Label>
   396→                        <Input
   397→                          id="phone"
   398→                          value={form.phone}
   399→                          onChange={(e) => updateField('phone', e.target.value)}
   400→                          placeholder="08xxxxxxxxxx"
   401→                          className={`h-10 ${formErrors.phone ? 'border-red-500' : ''}`}
   402→                        />
   403→                        {formErrors.phone && (
   404→                          <p className="text-xs text-red-500">{formErrors.phone}</p>
   405→                        )}
   406→                      </div>
   407→                      <div className="space-y-1.5 sm:col-span-2">
   408→                        <Label htmlFor="address" className="text-sm font-medium">
   409→                          {t('checkout.addressField', lang)} <span className="text-red-500">*</span>
   410→                        </Label>
   411→                        <Input
   412→                          id="address"
   413→                          value={form.address}
   414→                          onChange={(e) => updateField('address', e.target.value)}
   415→                          placeholder={lang === 'id' ? 'Jl. Contoh No. 123, RT/RW, Kelurahan, Kecamatan' : 'Street, Ward, District'}
   416→                          className={`h-10 ${formErrors.address ? 'border-red-500' : ''}`}
   417→                        />
   418→                        {formErrors.address && (
   419→                          <p className="text-xs text-red-500">{formErrors.address}</p>
   420→                        )}
   421→                      </div>
   422→                      <div className="space-y-1.5">
   423→                        <Label htmlFor="city" className="text-sm font-medium">
   424→                          {t('checkout.city', lang)} <span className="text-red-500">*</span>
   425→                        </Label>
   426→                        <Input
   427→                          id="city"
   428→                          value={form.city}
   429→                          onChange={(e) => updateField('city', e.target.value)}
   430→                          placeholder={lang === 'id' ? 'Jakarta Selatan' : 'South Jakarta'}
   431→                          className={`h-10 ${formErrors.city ? 'border-red-500' : ''}`}

   434→                          <p className="text-xs text-red-500">{formErrors.city}</p>

   436→                      </div>
   437→                      <div className="space-y-1.5">
   438→                        <Label htmlFor="postal" className="text-sm font-medium">
   439→                          {t('checkout.postal', lang)} <span className="text-red-500">*</span>
   440→                        </Label>
   441→                        <Input
   442→                          id="postal"
   443→                          value={form.postalCode}
   444→                          onChange={(e) => updateField('postalCode', e.target.value)}
   445→                          placeholder="12345"
   446→                          className={`h-10 ${formErrors.postalCode ? 'border-red-500' : ''}`}
   447→                        />
   448→                        {formErrors.postalCode && (

   451→                      </div>
   452→                    </div>
   453→                    <div className="mt-6 flex justify-end">

   456→                        className="bg-[#D4AF37] font-semibold text-white hover:bg-[#B8960C]"

   459→                        <ChevronRight className="ml-1 h-4 w-4" />
   460→                      </Button>
   461→                    </div>
   462→                  </CardContent>
   463→                </Card>
   464→              </motion.div>

   466→          </AnimatePresence>

   468→          {/* Step 1: Expedition & Payment */}
   469→          <AnimatePresence mode="wait">

   471→              <motion.div
   472→                key="step-shipping"
   473→                ref={shippingRef}

   476→                exit={{ opacity: 0, x: -20 }}
   477→                transition={{ duration: 0.3 }}
   478→                className="space-y-6"
   479→              >

   482→                  onClick={() => goToStep(0)}

   484→                  type="button"

   487→                  {t('general.back', lang)}

   501→                    </div>
   502→                    <RadioGroup value={expedition} onValueChange={setExpedition} className="grid gap-3">
   503→                      {EXPEDITIONS.map((exp) => {
   504→                        const isSelected = expedition === exp.id;
   505→                        return (
   506→                          <Label
   507→                            key={exp.id}
