   485→   RECOMMENDED FOR YOU SECTION (Buku Untukmu)

   487→function RecommendedSection({
   488→  locale,
   489→  books,
   490→  setPage,
   491→}: {
   492→  locale: 'id' | 'en';
   493→  books: Book[];
   494→  setPage: (page: Page, params?: Record<string, string>) => void;
   495→}) {
   496→  const { recentlyViewed } = useAppStore();
   497→  const ref = useRef<HTMLDivElement>(null);
   498→  const scrollRef = useRef<HTMLDivElement>(null);
   499→  const inView = useInView(ref, { once: true, margin: '-100px' });
   500→
   501→  const [recommended, setRecommended] = useState<Book[]>([]);
   502→  const [loading, setLoading] = useState(true);
   503→  const [error, setError] = useState(false);
   504→  const [refreshing, setRefreshing] = useState(false);
   505→  const [basedOnCategory, setBasedOnCategory] = useState(false);
   506→
   507→  const fetchRecommendations = useCallback(async (showRefresh = false) => {
   508→    if (showRefresh) setRefreshing(true);
   509→    else setLoading(true);
   510→    setError(false);
   511→
   512→    try {
   513→      // Determine the most recent book's categoryId from store
   514→      let categoryId: string | null = null;
   515→      if (recentlyViewed.length > 0) {
   516→        const mostRecentId = recentlyViewed[0];
   517→        const recentBook = books.find((b) => b.id === mostRecentId);
   518→        if (recentBook?.categoryId) {
   519→          categoryId = recentBook.categoryId;
   520→        }
   521→      }
   522→
   523→      const params = new URLSearchParams();
   524→      if (categoryId) {
   525→        params.set('categoryId', categoryId);
   526→      }
   527→      // Exclude recently viewed books from recommendations
   528→      if (recentlyViewed.length > 0) {
   529→        params.set('exclude', recentlyViewed.slice(0, 10).join(','));
   530→      }
   531→      params.set('limit', '8');
   532→
   533→      const res = await fetch(`/api/books/recommendations?${params.toString()}`);
   534→      if (!res.ok) throw new Error();
   535→
   536→      const data = await res.json();
   537→      setRecommended(data.books || []);
   538→      setBasedOnCategory(!!categoryId);
   539→    } catch {
   540→      setError(true);
   541→    } finally {
   542→      setLoading(false);
   543→      setRefreshing(false);
   544→    }
   545→  }, [recentlyViewed, books]);
   546→
   547→  useEffect(() => {
   548→    fetchRecommendations();
   549→  }, [fetchRecommendations]);
   550→
   551→  const handleRefresh = () => {
   552→    fetchRecommendations(true);
   553→    toast.success(t('recommend.refreshed', locale));
   554→  };
   555→
   556→  const scroll = (dir: 'left' | 'right') => {
   557→    if (!scrollRef.current) return;
   558→    scrollRef.current.scrollBy({
   559→      left: dir === 'left' ? -340 : 340,
   560→      behavior: 'smooth',
   561→    });
   562→  };
   563→
   564→  // Don't render if empty after loading (but always render during loading to keep the section visible)
   565→  if (!loading && !error && recommended.length === 0) return null;
   566→
   567→  return (
   568→    <section ref={ref} className="py-16 sm:py-20">
   569→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   570→        {/* Section header with Sparkles + refresh */}
   571→        <motion.div
   572→          initial="hidden"
   573→          animate={inView ? 'visible' : 'hidden'}
   574→          variants={fadeUp}
   575→          custom={0}
   576→          className="flex flex-col sm:flex-row sm:items-end sm:justify-between gap-4 mb-8"
   577→        >
   578→          <div>
   579→            <div className="flex items-center gap-3 mb-2">
   580→              <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-[#D4AF37]/10 border border-[#D4AF37]/20">
   581→                <Sparkles className="h-5 w-5 text-[#D4AF37]" />
   582→              </div>
   583→              <h2 className="font-heading text-2xl sm:text-3xl font-bold tracking-tight">
   584→                {t('recommend.title', locale)}
   585→              </h2>
   586→            </div>
   587→            <p className="text-muted-foreground text-sm sm:text-base max-w-md">
   588→              {t('recommend.subtitle', locale)}
   589→            </p>
   590→            {basedOnCategory && (
   591→              <Badge
   592→                variant="outline"
   593→                className="mt-2 border-[#D4AF37]/30 text-[#D4AF37] bg-[#D4AF37]/5 text-xs"
   594→              >
   595→                {t('recommend.basedOn', locale)}: {t('recommend.basedOnCategory', locale)}
   596→              </Badge>
   597→            )}
   598→          </div>
   599→
   600→          <Button
   601→            variant="outline"
   602→            size="sm"
   603→            onClick={handleRefresh}
   604→            disabled={refreshing}
   605→            className="border-[#D4AF37]/30 text-[#D4AF37] hover:bg-[#D4AF37] hover:text-white transition-all shrink-0 h-9 px-4"
   606→          >
   607→            <RefreshCw className={`mr-2 h-3.5 w-3.5 ${refreshing ? 'animate-spin' : ''}`} />
   608→            {t('recommend.refresh', locale)}
   609→          </Button>
   610→        </motion.div>
   611→      </div>
   612→
   613→      {/* Loading skeleton */}
   614→      {loading && (
   615→        <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   616→          <div className="flex gap-5 overflow-hidden">
   617→            {Array.from({ length: 6 }).map((_, i) => (
   618→              <div key={i} className="shrink-0 w-[200px] sm:w-[220px]">
   619→                <div className="aspect-[2/3] rounded-xl bg-muted animate-pulse" />
   620→                <div className="mt-3 h-4 w-3/4 rounded bg-muted animate-pulse" />
   621→                <div className="mt-2 h-3 w-1/2 rounded bg-muted animate-pulse" />
   622→              </div>
   623→            ))}
   624→          </div>
   625→        </div>
   626→      )}
   627→
   628→      {/* Error state */}
   629→      {error && !loading && (
   630→        <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   631→          <div className="flex flex-col items-center justify-center py-12 text-center">
   632→            <div className="flex h-16 w-16 items-center justify-center rounded-full bg-muted mb-4">
   633→              <Sparkles className="h-7 w-7 text-muted-foreground" />
   634→            </div>
   635→            <p className="text-lg font-semibold mb-1">{t('recommend.empty', locale)}</p>
   636→            <p className="text-sm text-muted-foreground mb-4 max-w-sm">
   637→              {t('recommend.error', locale)}
   638→            </p>
   639→            <Button
   640→              variant="outline"
   641→              onClick={handleRefresh}
   642→              className="border-[#D4AF37]/30 text-[#D4AF37] hover:bg-[#D4AF37] hover:text-white"
   643→            >
   644→              {t('recommend.retry', locale)}
   645→            </Button>
   646→          </div>
   647→        </div>
   648→      )}
   649→
   650→      {/* Book row */}
   651→      {!loading && !error && recommended.length > 0 && (
   652→        <div className="relative group/row">
   653→          {/* Left arrow */}
   654→          <button
   655→            onClick={() => scroll('left')}
   656→            className="absolute left-0 top-1/2 -translate-y-1/2 z-10 hidden lg:flex h-10 w-10 items-center justify-center rounded-full bg-white/90 shadow-lg border border-[#D4AF37]/20 text-[#D4AF37] hover:bg-[#D4AF37] hover:text-white transition-all opacity-0 group-hover/row:opacity-100"
   657→            aria-label="Scroll left"
   658→          >
   659→            <ChevronLeft className="h-5 w-5" />
