    93→function HorizontalBookRow({
    94→  books,
    95→  title,
    96→  subtitle,
    97→  viewAllLabel,
    98→  onViewAll,
    99→}: {
   100→  books: Book[];
   101→  title: string;
   102→  subtitle: string;
   103→  viewAllLabel: string;
   104→  onViewAll: () => void;
   105→}) {
   106→  const scrollRef = useRef<HTMLDivElement>(null);
   107→
   108→  const scroll = (dir: 'left' | 'right') => {
   109→    if (!scrollRef.current) return;
   110→    const amount = 340;
   111→    scrollRef.current.scrollBy({
   112→      left: dir === 'left' ? -amount : amount,
   113→      behavior: 'smooth',
   114→    });
   115→  };
   116→
   117→  if (books.length === 0) return null;
   118→
   119→  return (
   120→    <section className="py-16 sm:py-20">
   121→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   122→        <SectionHeader
   123→          title={title}
   124→          subtitle={subtitle}
   125→          action={{ label: viewAllLabel, onClick: onViewAll }}
   126→        />
   127→      </div>
   128→
   129→      <div className="relative group/row">
   130→        {/* Left arrow */}
   131→        <button
   132→          onClick={() => scroll('left')}
   133→          className="absolute left-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   134→          aria-label="Scroll left"
   135→        >
   136→          <ChevronLeft className="h-5 w-5" />
   137→        </button>
   138→        {/* Right arrow */}
   139→        <button
   140→          onClick={() => scroll('right')}
   141→          className="absolute right-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   142→          aria-label="Scroll right"
   143→        >
   144→          <ChevronRight className="h-5 w-5" />
   145→        </button>
   146→
   147→        <div
   148→          ref={scrollRef}
   149→          className="flex gap-5 overflow-x-auto scrollbar-hide px-4 sm:px-6 lg:px-8 pb-4 snap-x snap-mandatory"
   150→        >
   151→          {books.map((book, i) => (
