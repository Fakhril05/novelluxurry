    65→    65→function estimateReadTime(content: string | null): number {
    66→    66→  if (!content) return 1;
    67→    67→  const text = content.replace(/<[^>]*>/g, '');
    68→    68→  const wordCount = text.split(/\s+/).filter(Boolean).length;
    69→    69→  return Math.max(1, Math.ceil(wordCount / 200));
    70→    70→}
