    67→    67→function StarSelector({

    82→    82→          onMouseEnter={() => setHovered(star)}
    83→    83→          onMouseLeave={() => setHovered(0)}

    86→    86→          aria-label={`${star} stars`}
    87→    87→        >
    88→    88→          <Star
    89→    89→            className={`h-5 w-5 transition-colors ${
    90→    90→              star <= (hovered || value)
    91→    91→                ? 'fill-[#D4AF37] text-[#D4AF37]'
    92→    92→                : 'text-muted-foreground/30'
    93→    93→            }`}
    94→    94→          />
    95→    95→        </button>
    96→    96→      ))}

    98→    98→        <span className="ml-1 text-xs text-muted-foreground">{value}+</span>

   100→   100→    </div>

   104→   104→function FilterSidebar({
   105→   105→  categories,

   107→   107→  onFilterChange,
   108→   108→  onReset,
   109→   109→  locale,
   110→   110→}: {
   111→   111→  categories: Category[];
   112→   112→  filters: FilterState;
   113→   113→  onFilterChange: (filters: Record<string, unknown>) => void;
   114→   114→  onReset: () => void;
   115→   115→  locale: 'id' | 'en';
   116→   116→}) {
   117→   117→  const [localPriceMin, setLocalPriceMin] = useState(
   118→   118→    filters.minPrice ? String(filters.minPrice) : ''
   119→   119→  );

   121→   121→    filters.maxPrice ? String(filters.maxPrice) : ''
   122→   122→  );

   125→   125→    const min = localPriceMin ? parseInt(localPriceMin, 10) : undefined;
   126→   126→    const max = localPriceMax ? parseInt(localPriceMax, 10) : undefined;

   128→   128→  };
   129→   129→
   130→   130→  const selectedGenres = filters.genre
   131→   131→    ? filters.genre.split(',').filter(Boolean)
   132→   132→    : [];
   133→   133→
   134→   134→  const toggleGenre = (slug: string) => {
   135→   135→    const updated = selectedGenres.includes(slug)
   136→   136→      ? selectedGenres.filter((g) => g !== slug)
   137→   137→      : [...selectedGenres, slug];
   138→   138→    onFilterChange({ genre: updated.length > 0 ? updated.join(',') : undefined });
   139→   139→  };
   140→   140→
   141→   141→  return (
   142→   142→    <div className="space-y-6">
   143→   143→      {/* Sort */}
   144→   144→      <div>
   145→   145→        <h3 className="font-heading text-sm font-semibold uppercase tracking-wider mb-3 text-foreground">
   146→   146→          {t('catalog.sort', locale)}
   147→   147→        </h3>
   148→   148→        <div className="space-y-2">
   149→   149→          {SORT_OPTIONS.map((opt) => (
   150→   150→            <button
   151→   151→              key={opt.value}
   152→   152→              type="button"
   153→   153→              onClick={() => onFilterChange({ sort: opt.value })}
   154→   154→              className={`block w-full text-left text-sm px-3 py-2 rounded-lg transition-colors ${
   155→   155→                filters.sort === opt.value
   156→   156→                  ? 'bg-[#D4AF37]/10 text-[#D4AF37] font-medium'
   157→   157→                  : 'text-muted-foreground hover:bg-muted hover:text-foreground'
   158→   158→              }`}
   159→   159→            >
   160→   160→              {t(opt.labelKey, locale)}
   161→   161→            </button>
   162→   162→          ))}
   163→   163→        </div>
   164→   164→      </div>
   165→   165→
   166→   166→      {/* Divider */}
   167→   167→      <div className="border-t border-border" />
   168→   168→
   169→   169→      {/* Genre */}
   170→   170→      <div>
   171→   171→        <h3 className="font-heading text-sm font-semibold uppercase tracking-wider mb-3 text-foreground">
   172→   172→          {t('catalog.filter.genre', locale)}
   173→   173→        </h3>
   174→   174→        <ScrollArea className="max-h-56 overflow-y-auto">
   175→   175→          <div className="space-y-2.5 pr-1">
   176→   176→            {categories.map((cat) => (
   177→   177→              <label
   178→   178→                key={cat.id}
   179→   179→                className="flex items-center gap-2.5 cursor-pointer group"
   180→   180→              >
   181→   181→                <Checkbox
   182→   182→                  checked={selectedGenres.includes(cat.slug)}
   183→   183→                  onCheckedChange={() => toggleGenre(cat.slug)}
   184→   184→                  className="data-[state=checked]:bg-[#D4AF37] data-[state=checked]:border-[#D4AF37]"
   185→   185→                />
   186→   186→                <span className="text-sm text-muted-foreground group-hover:text-foreground transition-colors flex-1">
   187→   187→                  {locale === 'en' && cat.nameEn ? cat.nameEn : cat.name}
   188→   188→                </span>
   189→   189→                <span className="text-xs text-muted-foreground/60">
   190→   190→                  {cat._count?.books || 0}
   191→   191→                </span>
   192→   192→              </label>
   193→   193→            ))}
   194→   194→          </div>
   195→   195→        </ScrollArea>
   196→   196→      </div>
   197→   197→
   198→   198→      {/* Divider */}
   199→   199→      <div className="border-t border-border" />
   200→   200→

   204→   204→          {t('catalog.filter.price', locale)}
   205→   205→        </h3>

   207→   207→          <Input
   208→   208→            type="number"
   209→   209→            placeholder="Min"
   210→   210→            value={localPriceMin}
   211→   211→            onChange={(e) => setLocalPriceMin(e.target.value)}
   212→   212→            className="h-9 text-sm"
   213→   213→            onKeyDown={(e) => e.key === 'Enter' && handlePriceSubmit()}
   214→   214→          />
   215→   215→          <span className="text-muted-foreground text-sm">—</span>
   216→   216→          <Input
   217→   217→            type="number"
   218→   218→            placeholder="Max"
   219→   219→            value={localPriceMax}
   220→   220→            onChange={(e) => setLocalPriceMax(e.target.value)}
   221→   221→            className="h-9 text-sm"
   222→   222→            onKeyDown={(e) => e.key === 'Enter' && handlePriceSubmit()}
   223→   223→          />
   224→   224→        </div>
   225→   225→        <Button
   226→   226→          variant="outline"
   227→   227→          size="sm"
   228→   228→          onClick={handlePriceSubmit}
   229→   229→          className="mt-2 w-full h-8 text-xs border-[#D4AF37]/30 text-[#D4AF37] hover:bg-[#D4AF37]/10 hover:text-[#D4AF37]"
   230→   230→        >
   231→   231→          {t('catalog.filter.apply', locale)}
   232→   232→        </Button>
   233→   233→      </div>
   234→   234→
   235→   235→      {/* Divider */}
   236→   236→      <div className="border-t border-border" />
   237→   237→
   238→   238→      {/* Rating */}
   239→   239→      <div>
   240→   240→        <h3 className="font-heading text-sm font-semibold uppercase tracking-wider mb-3 text-foreground">
   241→   241→          {t('catalog.filter.rating', locale)}
   242→   242→        </h3>
   243→   243→        <StarSelector
   244→   244→          value={filters.minRating || 0}
   245→   245→          onChange={(val) => onFilterChange({ minRating: val || undefined })}

   247→   247→      </div>
   248→   248→
   249→   249→      {/* Reset */}

   254→   254→          onClick={onReset}
   255→   255→          className="w-full text-muted-foreground hover:text-foreground"
   256→   256→        >
   257→   257→          {t('catalog.filter.reset', locale)}
   258→   258→        </Button>
   259→   259→      </div>
   260→   260→    </div>
   261→   261→  );
   262→   262→}

   307→   307→export default function CatalogPage() {
   308→   308→  const { locale, pageParams, setPage, filters, setFilters, resetFilters } =
   309→   309→    useAppStore();
   310→   310→
   311→   311→  const [books, setBooks] = useState<Book[]>([]);
   312→   312→  const [categories, setCategories] = useState<Category[]>([]);
   313→   313→  const [total, setTotal] = useState(0);
   314→   314→  const [currentPage, setCurrentPage] = useState(1);
   315→   315→  const [loading, setLoading] = useState(true);
   316→   316→  const [categoriesLoading, setCategoriesLoading] = useState(true);
   317→   317→  const [searchInput, setSearchInput] = useState('');
   318→   318→  const [sheetOpen, setSheetOpen] = useState(false);
   319→   319→
   320→   320→  // Sync genre from pageParams on mount
   321→   321→  useEffect(() => {
   322→   322→    if (pageParams.genre && !filters.genre) {
   323→   323→      setFilters({ genre: pageParams.genre });
   324→   324→    }
   325→   325→    if (pageParams.search) {
   326→   326→      setSearchInput(pageParams.search);
   327→   327→    }
   328→   328→  }, []);
   329→   329→
   330→   330→  // Fetch categories
   331→   331→  useEffect(() => {
   332→   332→    let cancelled = false;
   333→   333→    const fetchCategories = async () => {
   334→   334→      try {
   335→   335→        const res = await fetch('/api/categories');
   336→   336→        if (res.ok && !cancelled) {
   337→   337→          const data = await res.json();
   338→   338→          setCategories(Array.isArray(data) ? data : data.categories || []);
   339→   339→        }
   340→   340→      } catch {
   341→   341→        // silently fail
   342→   342→      } finally {
   343→   343→        if (!cancelled) setCategoriesLoading(false);
   344→   344→      }
   345→   345→    };
   346→   346→    fetchCategories();
   347→   347→    return () => {
   348→   348→      cancelled = true;
   349→   349→    };
   350→   350→  }, []);
   351→   351→
   352→   352→  // Reset to page 1 when filters change (except page itself)
   353→   353→  useEffect(() => {
   354→   354→    setCurrentPage(1);
   355→   355→  }, [filters.genre, filters.minPrice, filters.maxPrice, filters.minRating, filters.sort, pageParams.search]);
   356→   356→
   357→   357→  // Fetch books
   358→   358→  useEffect(() => {
   359→   359→    let cancelled = false;
   360→   360→    const doFetch = async () => {
   361→   361→      setLoading(true);
   362→   362→      try {
   363→   363→        const params = new URLSearchParams();
   364→   364→        params.set('limit', String(LIMIT));
   365→   365→        params.set('page', String(currentPage));
   366→   366→
   367→   367→        if (filters.genre) params.set('genre', filters.genre);
   368→   368→        if (filters.minPrice !== undefined && filters.minPrice > 0)
   369→   369→          params.set('minPrice', String(filters.minPrice));
   370→   370→        if (filters.maxPrice !== undefined && filters.maxPrice > 0)
   371→   371→          params.set('maxPrice', String(filters.maxPrice));
   372→   372→        if (filters.minRating !== undefined && filters.minRating > 0)
   373→   373→          params.set('minRating', String(filters.minRating));
   374→   374→        if (filters.sort) params.set('sort', filters.sort);
   375→   375→        if (pageParams.search) params.set('search', pageParams.search);
   376→   376→
   377→   377→        const res = await fetch(`/api/books?${params.toString()}`);
   378→   378→        if (res.ok && !cancelled) {
   379→   379→          const data: BooksResponse = await res.json();
   380→   380→          setBooks(data.books || []);
   381→   381→          setTotal(data.total || 0);
   382→   382→        }
   383→   383→      } catch {
   384→   384→        // silently fail
   385→   385→      } finally {
   386→   386→        if (!cancelled) setLoading(false);
   387→   387→      }
   388→   388→    };
   389→   389→    doFetch();
   390→   390→    return () => {
   391→   391→      cancelled = true;
   392→   392→    };
   393→   393→  }, [filters, currentPage, pageParams.search]);
   394→   394→
   395→   395→  const totalPages = Math.max(1, Math.ceil(total / LIMIT));
   396→   396→
   397→   397→  const handleFilterChange = useCallback(
   398→   398→    (newFilters: Record<string, unknown>) => {
   399→   399→      setFilters(newFilters as Partial<FilterState>);
   400→   400→    },
   401→   401→    [setFilters]
   402→   402→  );
   403→   403→
   404→   404→  const handleReset = useCallback(() => {
   405→   405→    resetFilters();
   406→   406→    setSearchInput('');
   407→   407→    setCurrentPage(1);
   408→   408→    setPage('catalog', {});
   409→   409→  }, [resetFilters, setPage]);
   410→   410→
   411→   411→  const handleSearch = useCallback(
   412→   412→    (e: React.FormEvent) => {
   413→   413→      e.preventDefault();
   414→   414→      setPage('catalog', { ...pageParams, search: searchInput.trim() });
   415→   415→      setCurrentPage(1);
   416→   416→    },
   417→   417→    [searchInput, pageParams, setPage]
   418→   418→  );
   419→   419→
   420→   420→  // Active filter badges
   421→   421→  const activeFilters = useMemo(() => {
   422→   422→    const items: { key: string; label: string; clear: () => void }[] = [];
   423→   423→
   424→   424→    if (filters.genre) {
   425→   425→      const slugs = filters.genre.split(',').filter(Boolean);
   426→   426→      slugs.forEach((slug) => {
   427→   427→        const cat = categories.find((c) => c.slug === slug);
   428→   428→        const name = cat
   429→   429→          ? locale === 'en' && cat.nameEn
   430→   430→            ? cat.nameEn
   431→   431→            : cat.name
   432→   432→          : slug;
   433→   433→        items.push({
   434→   434→          key: `genre-${slug}`,
   435→   435→          label: name,
   436→   436→          clear: () => {
   437→   437→            const updated = slugs.filter((s) => s !== slug);
   438→   438→            setFilters({ genre: updated.length > 0 ? updated.join(',') : undefined });
   439→   439→          },
   440→   440→        });
   441→   441→      });
   442→   442→    }
   443→   443→
   444→   444→    if (filters.minPrice !== undefined && filters.minPrice > 0) {
   445→   445→      items.push({
   446→   446→        key: 'minPrice',
   447→   447→        label: `Min ${formatPrice(filters.minPrice, locale)}`,
   448→   448→        clear: () => setFilters({ minPrice: undefined }),
   449→   449→      });
   450→   450→    }
   451→   451→
   452→   452→    if (filters.maxPrice !== undefined && filters.maxPrice > 0) {
   453→   453→      items.push({
   454→   454→        key: 'maxPrice',
   455→   455→        label: `Max ${formatPrice(filters.maxPrice, locale)}`,
   456→   456→        clear: () => setFilters({ maxPrice: undefined }),
   457→   457→      });
   458→   458→    }
   459→   459→
   460→   460→    if (filters.minRating !== undefined && filters.minRating > 0) {
   461→   461→      items.push({
   462→   462→        key: 'minRating',
   463→   463→        label: `${filters.minRating}+ \u2605`,
   464→   464→        clear: () => setFilters({ minRating: undefined }),
   465→   465→      });
   466→   466→    }
   467→   467→
   468→   468→    if (pageParams.search) {
   469→   469→      items.push({
   470→   470→        key: 'search',
   471→   471→        label: `"${pageParams.search}"`,
   472→   472→        clear: () => {
   473→   473→          setSearchInput('');
   474→   474→          setPage('catalog', { ...pageParams, search: '' });
   475→   475→        },
   476→   476→      });
   477→   477→    }
   478→   478→
   479→   479→    return items;
   480→   480→  }, [filters, categories, locale, pageParams, setFilters, setPage]);
   481→   481→
   482→   482→  // Breadcrumb: if single genre filtered, show the category name
   483→   483→  const genreCategory = useMemo(() => {
   484→   484→    if (!filters.genre) return null;
   485→   485→    const slugs = filters.genre.split(',').filter(Boolean);
   486→   486→    if (slugs.length === 1) {
   487→   487→      return categories.find((c) => c.slug === slugs[0]) || null;
   488→   488→    }
   489→   489→    return null;
