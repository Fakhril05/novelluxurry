    68→function estimateReadTime(content: string | null): number {
    69→  if (!content) return 1;
    70→  const text = content.replace(/<[^>]*>/g, '');
    71→  const wordCount = text.split(/\s+/).filter(Boolean).length;
    72→  return Math.max(1, Math.ceil(wordCount / 200));
    73→}

    75→function formatDate(dateStr: string, locale: Locale): string {
    76→  const date = new Date(dateStr);
    77→  return date.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {
    78→    year: 'numeric',
    79→    month: 'long',
    80→    day: 'numeric',
    81→  });
    82→}

    84→function getLocalizedField<T extends string | null>(
    85→  idValue: T,
    86→  enValue: T,
    87→  locale: Locale
    88→): T {
    89→  if (locale === 'en' && enValue) return enValue;
    90→  return idValue;
    91→}

    94→  return author

   103→/* Gradient color palettes for category-based placeholders */
   104→const CATEGORY_GRADIENTS: string[] = [

   106→  'linear-gradient(135deg, #2d1b2e 0%, #44203a 50%, #6b2d5b 100%)',
   107→  'linear-gradient(135deg, #1b2d1b 0%, #1e3a20 50%, #2d5a3f 100%)',

   111→
   112→function getCategoryGradient(slug: string): string {
   113→  let hash = 0;
   114→  for (let i = 0; i < slug.length; i++) {
   115→    hash = slug.charCodeAt(i) + ((hash << 5) - hash);
   116→  }
   117→  return CATEGORY_GRADIENTS[Math.abs(hash) % CATEGORY_GRADIENTS.length];
   118→}

   488→/* ─── Main Component ─── */
   489→
   490→export default function BlogPage() {
   491→  const { locale, pageParams, setPage } = useAppStore();
   492→  const [blogs, setBlogs] = useState<BlogPost[]>([]);
   493→  const [total, setTotal] = useState(0);
   494→  const [currentBlog, setCurrentBlog] = useState<BlogPost | null>(null);
   495→  const [loading, setLoading] = useState(true);
   496→  const [loadingMore, setLoadingMore] = useState(false);
   497→  const [detailLoading, setDetailLoading] = useState(false);
   498→  const [visibleCount, setVisibleCount] = useState(POSTS_PER_PAGE);
   499→  const [selectedCategory, setSelectedCategory] = useState('all');
   500→
   501→  const slug = pageParams.slug as string | undefined;
   502→
   503→  const fetchBlogs = useCallback(async () => {
   504→    try {
   505→      const res = await fetch('/api/blogs');
   506→      if (res.ok) {
   507→        const data = await res.json();
   508→        setBlogs(data.blogs || data);
   509→        setTotal(data.total ?? data.length ?? 0);
   510→      }
   511→    } catch {
   512→      // silently fail
   513→    } finally {
   514→      setLoading(false);
   515→    }
   516→  }, []);
   517→
   518→  useEffect(() => {
   519→    fetchBlogs();
   520→  }, [fetchBlogs]);
   521→
   522→  useEffect(() => {
   523→    if (slug) {
   524→      setDetailLoading(true);
   525→      async function fetchBlog() {
   526→        try {
   527→          const res = await fetch(`/api/blogs/${slug}`);
   528→          if (res.ok) {
   529→            const data = await res.json();
   530→            setCurrentBlog(data);
   531→          } else {
   532→            setCurrentBlog(null);
   533→          }
   534→        } catch {
   535→          setCurrentBlog(null);
   536→        } finally {
   537→          setDetailLoading(false);
   538→        }
   539→      }
   540→      fetchBlog();
   541→    } else {
   542→      setCurrentBlog(null);
   543→    }
   544→  }, [slug]);
   545→
   546→  // Collect all unique categories
   547→  const allCategories = useMemo(() => {
   548→    const cats = new Set<string>();
   549→    blogs.forEach((b) => {
   550→      const cat = getLocalizedField(b.category, b.categoryEn, locale);
   551→      if (cat) cats.add(cat);
   552→    });
   553→    return Array.from(cats);
   554→  }, [blogs, locale]);
   555→
   556→  // Filter blogs by category
   557→  const filteredBlogs = useMemo(() => {
   558→    if (selectedCategory === 'all') return blogs;
   559→    return blogs.filter((b) => {
   560→      const cat = getLocalizedField(b.category, b.categoryEn, locale);
   561→      return cat === selectedCategory;
   562→    });
   563→  }, [blogs, selectedCategory, locale]);
   564→
   565→  const hasMore = visibleCount < filteredBlogs.length;
   566→
   567→  const handleLoadMore = () => {
   568→    setLoadingMore(true);
   569→    setTimeout(() => {
   570→      setVisibleCount((prev) => prev + POSTS_PER_PAGE);
   571→      setLoadingMore(false);
   572→    }, 400);
   573→  };
   574→
   575→  const visibleBlogs = useMemo(
   576→    () => filteredBlogs.slice(0, visibleCount),
   577→    [filteredBlogs, visibleCount]
   578→  );
   579→
   580→  const featuredBlog = visibleBlogs[0];
   581→  const restBlogs = visibleBlogs.slice(1);
