   397→   397→function BooksTab({
   398→   398→  locale,
   399→   399→  books,
   400→   400→  categories,
   401→   401→  onRefresh,
   402→   402→}: {
   403→   403→  locale: string;
   404→   404→  books: Book[];
   405→   405→  categories: Category[];
   406→   406→  onRefresh: () => void;
   407→   407→}) {
   408→   408→  const [dialogOpen, setDialogOpen] = useState(false);
   409→   409→  const [editingBook, setEditingBook] = useState<Book | null>(null);
   410→   410→  const [form, setForm] = useState<BookFormData>(EMPTY_BOOK_FORM);
   411→   411→  const [saving, setSaving] = useState(false);
   412→   412→  const [deletingId, setDeletingId] = useState<string | null>(null);
   413→   413→
   414→   414→  const openAddDialog = useCallback(() => {
   415→   415→    setEditingBook(null);
   416→   416→    setForm(EMPTY_BOOK_FORM);
   417→   417→    setDialogOpen(true);
   418→   418→  }, []);
   419→   419→
   420→   420→  const openEditDialog = useCallback((book: Book) => {
   421→   421→    setEditingBook(book);
   422→   422→    setForm({
   423→   423→      title: book.title,
   424→   424→      author: book.author,
   425→   425→      authorBio: book.authorBio || '',
   426→   426→      isbn: book.isbn || '',
   427→   427→      synopsis: book.synopsis || '',
   428→   428→      coverImage: book.coverImage,
   429→   429→      price: String(book.price),
   430→   430→      discountPrice: book.discountPrice ? String(book.discountPrice) : '',
   431→   431→      stock: String(book.stock),
   432→   432→      format: book.format,
   433→   433→      pages: book.pages ? String(book.pages) : '',
   434→   434→      publisher: book.publisher || '',
   435→   435→      language: book.language,
   436→   436→      publishedYear: book.publishedYear ? String(book.publishedYear) : '',
   437→   437→      categoryId: book.categoryId || '',
   438→   438→      isBestSeller: book.isBestSeller,
   439→   439→      isNewArrival: book.isNewArrival,
   440→   440→      isFeatured: book.isFeatured,
   441→   441→    });
   442→   442→    setDialogOpen(true);
   443→   443→  }, []);
   444→   444→
   445→   445→  const handleSave = useCallback(async () => {
   446→   446→    if (!form.title || !form.author || !form.price) {
   447→   447→      toast.error(locale === 'id' ? 'Judul, penulis, dan harga wajib diisi' : 'Title, author, and price are required');
   448→   448→      return;
   449→   449→    }
   450→   450→    setSaving(true);
   451→   451→    try {
   452→   452→      const payload = {
   453→   453→        ...form,
   454→   454→        price: Number(form.price),
   455→   455→        discountPrice: form.discountPrice ? Number(form.discountPrice) : null,
   456→   456→        stock: Number(form.stock),
   457→   457→        pages: form.pages ? Number(form.pages) : null,
   458→   458→        publishedYear: form.publishedYear ? Number(form.publishedYear) : null,
   459→   459→      };
   460→   460→
   461→   461→      if (editingBook) {
   462→   462→        const res = await fetch(`/api/books?id=${editingBook.id}`, {
   463→   463→          method: 'PUT',
   464→   464→          headers: { 'Content-Type': 'application/json' },
   465→   465→          body: JSON.stringify(payload),
   466→   466→        });
   467→   467→        if (res.ok) {
   468→   468→          toast.success(locale === 'id' ? 'Buku berhasil diperbarui' : 'Book updated successfully');
   469→   469→          setDialogOpen(false);
   470→   470→          onRefresh();
   471→   471→        } else {
   472→   472→          toast.error(locale === 'id' ? 'Gagal memperbarui buku' : 'Failed to update book');
   473→   473→        }
   474→   474→      } else {
   475→   475→        const res = await fetch('/api/books', {
   476→   476→          method: 'POST',
   477→   477→          headers: { 'Content-Type': 'application/json' },
   478→   478→          body: JSON.stringify(payload),
   479→   479→        });
   480→   480→        if (res.ok) {
   481→   481→          toast.success(locale === 'id' ? 'Buku berhasil ditambahkan' : 'Book added successfully');
   482→   482→          setDialogOpen(false);
   483→   483→          onRefresh();
   484→   484→        } else {
   485→   485→          toast.error(locale === 'id' ? 'Gagal menambahkan buku' : 'Failed to add book');
   486→   486→        }
   487→   487→      }
