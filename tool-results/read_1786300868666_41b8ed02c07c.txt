    65→function estimateReadTime(content: string | null): number {
    66→  if (!content) return 1;
    67→  const text = content.replace(/<[^>]*>/g, '');
    68→  const wordCount = text.split(/\s+/).filter(Boolean).length;
    69→  return Math.max(1, Math.ceil(wordCount / 200));
    70→}

    72→function formatDate(dateStr: string, locale: Locale): string {
    73→  const date = new Date(dateStr);
    74→  return date.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {
    75→    year: 'numeric',
    76→    month: 'long',
    77→    day: 'numeric',
    78→  });
    79→}
    80→
    81→function getLocalizedField<T extends string | null>(
    82→  idValue: T,
    83→  enValue: T,
    84→  locale: Locale
    85→): T {
    86→  if (locale === 'en' && enValue) return enValue;
    87→  return idValue;
    88→}

    90→function getAuthorInitials(author: string): string {
    91→  return author
    92→    .split(' ')
    93→    .filter(Boolean)
    94→    .map((w) => w[0])
    95→    .slice(0, 2)
    96→    .join('')
    97→    .toUpperCase();
    98→}
