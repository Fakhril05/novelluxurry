    72→    72→/** Generate simulated rating distribution from average */
    73→    73→function getRatingDistribution(avgRating: number, totalReviews: number) {
    74→    74→  // Create a bell-curve-like distribution based on the average rating
    75→    75→  const base = totalReviews || 1;
    76→    76→  const dist: number[] = [0, 0, 0, 0, 0]; // 5★, 4★, 3★, 2★, 1★
    77→    77→  for (let i = 0; i < 5; i++) {
    78→    78→    const starVal = 5 - i;
    79→    79→    const diff = Math.abs(starVal - avgRating);
    80→    80→    dist[i] = Math.max(5, Math.round(base * Math.exp(-diff * diff * 0.5)));
    81→    81→  }
    82→    82→  // Normalize so sum matches totalReviews
    83→    83→  const sum = dist.reduce((a, b) => a + b, 0);
    84→    84→  for (let i = 0; i < 5; i++) {
    85→    85→    dist[i] = Math.round((dist[i] / sum) * base);
    86→    86→  }
    87→    87→  return dist;
    88→    88→}
