   110→}: {

   124→    setLocalPriceMax(priceMaxInput);

   128→    const min = localPriceMin ? parseInt(localPriceMin, 10) : undefined;
   129→    const max = localPriceMax ? parseInt(localPriceMax, 10) : undefined;
   130→    onFilterChange({ minPrice: min, maxPrice: max });
   131→  };

   133→  const selectedGenres = filters.genre
   134→    ? filters.genre.split(',').filter(Boolean)
   135→    : [];

   137→  const toggleGenre = (slug: string) => {
   138→    const updated = selectedGenres.includes(slug)
   139→      ? selectedGenres.filter((g) => g !== slug)
   140→      : [...selectedGenres, slug];
   141→    onFilterChange({ genre: updated.length > 0 ? updated.join(',') : undefined });
   142→  };

   145→    <div className="space-y-6">

   159→                  ? 'bg-[#D4AF37]/10 text-[#D4AF37] font-medium'

   202→      <div className="border-t border-border" />

   211→            type="number"

   226→          />
   227→        </div>
   228→        <Button
   229→          variant="outline"
   230→          size="sm"
   231→          onClick={handlePriceSubmit}
   232→          className="mt-2 w-full h-8 text-xs border-[#D4AF37]/30 text-[#D4AF37] hover:bg-[#D4AF37]/10 hover:text-[#D4AF37]"
