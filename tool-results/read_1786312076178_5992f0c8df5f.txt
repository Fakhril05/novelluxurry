    68→function StarSelector({
    69→  value,
    70→  onChange,
    71→}: {
    72→  value: number;
    73→  onChange: (val: number) => void;
    74→}) {
    75→  const [hovered, setHovered] = useState(0);
    76→
    77→  return (
    78→    <div className="flex items-center gap-0.5">
    79→      {[1, 2, 3, 4, 5].map((star) => (
    80→        <button
    81→          key={star}
    82→          type="button"
    83→          onMouseEnter={() => setHovered(star)}
    84→          onMouseLeave={() => setHovered(0)}
    85→          onClick={() => onChange(value === star ? 0 : star)}
    86→          className="p-0.5 transition-transform hover:scale-110 focus:outline-none focus-visible:ring-2 focus-visible:ring-[#D4AF37] rounded"
    87→          aria-label={`${star} stars`}
    88→        >
    89→          <Star
    90→            className={`h-5 w-5 transition-colors ${
    91→              star <= (hovered || value)
    92→                ? 'fill-[#D4AF37] text-[#D4AF37]'
    93→                : 'text-muted-foreground/30'
    94→            }`}
    95→          />
    96→        </button>
    97→      ))}
    98→      {value > 0 && (
    99→        <span className="ml-1 text-xs text-muted-foreground">{value}+</span>
   100→      )}
   101→    </div>
   102→  );
   103→}

   105→function FilterSidebar({
   106→  categories,
   107→  filters,
   108→  onFilterChange,
   109→  onReset,
   110→  locale,
   111→}: {
   112→  categories: Category[];
   113→  filters: FilterState;
   114→  onFilterChange: (filters: Record<string, unknown>) => void;
   115→  onReset: () => void;
   116→  locale: 'id' | 'en';
   117→}) {
   118→  const [localPriceMin, setLocalPriceMin] = useState(
   119→    filters.minPrice ? String(filters.minPrice) : ''
   120→  );
   121→  const [localPriceMax, setLocalPriceMax] = useState(
   122→    filters.maxPrice ? String(filters.maxPrice) : ''
   123→  );
   124→
   125→  const handlePriceSubmit = () => {
   126→    const min = localPriceMin ? parseInt(localPriceMin, 10) : undefined;
   127→    const max = localPriceMax ? parseInt(localPriceMax, 10) : undefined;
   128→    onFilterChange({ minPrice: min, maxPrice: max });
   129→  };
   130→
   131→  const selectedGenres = filters.genre
   132→    ? filters.genre.split(',').filter(Boolean)
   133→    : [];
   134→
   135→  const toggleGenre = (slug: string) => {
   136→    const updated = selectedGenres.includes(slug)
   137→      ? selectedGenres.filter((g) => g !== slug)
   138→      : [...selectedGenres, slug];
   139→    onFilterChange({ genre: updated.length > 0 ? updated.join(',') : undefined });
   140→  };
   141→
   142→  return (
   143→    <div className="space-y-6 border-t-2 border-[#D4AF37]/20 pt-6">
   144→      {/* Sort */}
   145→      <div>
   146→        <h3 className="font-heading text-sm font-semibold uppercase tracking-wider mb-3 text-foreground">
   147→          {t('catalog.sort', locale)}
   148→        </h3>
   149→        <div className="space-y-2">
   150→          {SORT_OPTIONS.map((opt) => (
   151→            <button
   152→              key={opt.value}
   153→              type="button"
   154→              onClick={() => onFilterChange({ sort: opt.value })}
   155→              className={`block w-full text-left text-sm px-3 py-2 rounded-lg transition-all duration-200 ${
   156→                filters.sort === opt.value
   157→                  ? 'bg-[#D4AF37]/10 text-[#D4AF37] font-medium border border-[#D4AF37]/30 shadow-sm shadow-[#D4AF37]/10'
   158→                  : 'text-muted-foreground hover:bg-muted hover:text-foreground border border-transparent'
   159→              }`}
   160→            >
   161→              {t(opt.labelKey, locale)}
   162→            </button>
   163→          ))}
   164→        </div>
   165→      </div>
   166→
   167→      {/* Divider */}
   168→      <div className="border-t border-border" />
   169→
   170→      {/* Genre */}
   171→      <div>
   172→        <h3 className="font-heading text-sm font-semibold uppercase tracking-wider mb-3 text-foreground">
   173→          {t('catalog.filter.genre', locale)}
   174→        </h3>
   175→        <ScrollArea className="max-h-56 overflow-y-auto">
   176→          <div className="space-y-2.5 pr-1">
   177→            {categories.map((cat) => (
   178→              <label
   179→                key={cat.id}
   180→                className="flex items-center gap-2.5 cursor-pointer group"
   181→              >
   182→                <Checkbox
   183→                  checked={selectedGenres.includes(cat.slug)}
   184→                  onCheckedChange={() => toggleGenre(cat.slug)}
   185→                  className="data-[state=checked]:bg-[#D4AF37] data-[state=checked]:border-[#D4AF37]"
   186→                />
   187→                <span className="text-sm text-muted-foreground group-hover:text-foreground transition-colors flex-1">
   188→                  {locale === 'en' && cat.nameEn ? cat.nameEn : cat.name}
   189→                </span>
   190→                <span className="text-xs text-muted-foreground/60">
   191→                  {cat._count?.books || 0}
   192→                </span>
   193→              </label>
   194→            ))}
   195→          </div>
   196→        </ScrollArea>
   197→      </div>
   198→
   199→      {/* Divider */}
   200→      <div className="border-t border-border" />
   201→
   202→      {/* Price Range */}
   203→      <div>
   204→        <h3 className="font-heading text-sm font-semibold uppercase tracking-wider mb-3 text-foreground">
   205→          {t('catalog.filter.price', locale)}
   206→        </h3>
   207→        <div className="flex items-center gap-2">
   208→          <Input
   209→            type="number"
   210→            placeholder="Min"
   211→            value={localPriceMin}
   212→            onChange={(e) => setLocalPriceMin(e.target.value)}
   213→            className="h-9 text-sm"
   214→            onKeyDown={(e) => e.key === 'Enter' && handlePriceSubmit()}
   215→          />
   216→          <span className="text-muted-foreground text-sm">—</span>
   217→          <Input
   218→            type="number"
   219→            placeholder="Max"
   220→            value={localPriceMax}
   221→            onChange={(e) => setLocalPriceMax(e.target.value)}
   222→            className="h-9 text-sm"
   223→            onKeyDown={(e) => e.key === 'Enter' && handlePriceSubmit()}
   224→          />
   225→        </div>
   226→        <Button
   227→          variant="outline"
   228→          size="sm"
   229→          onClick={handlePriceSubmit}
   230→          className="mt-2 w-full h-8 text-xs border-[#D4AF37]/30 text-[#D4AF37] hover:bg-[#D4AF37]/10 hover:text-[#D4AF37]"
   231→        >
   232→          {t('catalog.filter.apply', locale)}
   233→        </Button>
   234→      </div>
   235→
   236→      {/* Divider */}
   237→      <div className="border-t border-border" />
   238→
   239→      {/* Rating */}
   240→      <div>
   241→        <h3 className="font-heading text-sm font-semibold uppercase tracking-wider mb-3 text-foreground">
   242→          {t('catalog.filter.rating', locale)}
   243→        </h3>
   244→        <StarSelector
   245→          value={filters.minRating || 0}
   246→          onChange={(val) => onFilterChange({ minRating: val || undefined })}
   247→        />
   248→      </div>
   249→
   250→      {/* Reset */}
   251→      <div className="pt-2">
   252→        <Button
   253→          variant="ghost"
   254→          size="sm"
   255→          onClick={onReset}
   256→          className="w-full text-muted-foreground hover:text-foreground"
   257→        >
   258→          {t('catalog.filter.reset', locale)}
   259→        </Button>
   260→      </div>
   261→    </div>
   262→  );
   263→}

   349→export default function CatalogPage() {
   350→  const { locale, pageParams, setPage, filters, setFilters, resetFilters } =
   351→    useAppStore();
   352→
   353→  const [books, setBooks] = useState<Book[]>([]);
   354→  const [categories, setCategories] = useState<Category[]>([]);
   355→  const [total, setTotal] = useState(0);
   356→  const [currentPage, setCurrentPage] = useState(1);
   357→  const [loading, setLoading] = useState(true);
   358→  const [categoriesLoading, setCategoriesLoading] = useState(true);
   359→  const [searchInput, setSearchInput] = useState('');
   360→  const [sheetOpen, setSheetOpen] = useState(false);
   361→
   362→  // Sync genre from pageParams on mount
   363→  useEffect(() => {
   364→    if (pageParams.genre && !filters.genre) {
   365→      setFilters({ genre: pageParams.genre });
   366→    }
   367→    if (pageParams.search) {
   368→      setSearchInput(pageParams.search);
   369→    }
   370→  }, []);
   371→
   372→  // Fetch categories
   373→  useEffect(() => {
   374→    let cancelled = false;
   375→    const fetchCategories = async () => {
   376→      try {
   377→        const res = await fetch('/api/categories');
   378→        if (res.ok && !cancelled) {
   379→          const data = await res.json();
   380→          setCategories(Array.isArray(data) ? data : data.categories || []);
   381→        }
   382→      } catch {
   383→        // silently fail
   384→      } finally {
   385→        if (!cancelled) setCategoriesLoading(false);
   386→      }
   387→    };
   388→    fetchCategories();
   389→    return () => {
   390→      cancelled = true;
   391→    };
   392→  }, []);
   393→
   394→  // Reset to page 1 when filters change (except page itself)
   395→  useEffect(() => {
   396→    setCurrentPage(1);
   397→  }, [filters.genre, filters.minPrice, filters.maxPrice, filters.minRating, filters.sort, pageParams.search]);
   398→
   399→  // Fetch books
   400→  useEffect(() => {
   401→    let cancelled = false;
   402→    const doFetch = async () => {
   403→      setLoading(true);
   404→      try {
   405→        const params = new URLSearchParams();
   406→        params.set('limit', String(LIMIT));
   407→        params.set('page', String(currentPage));
   408→
   409→        if (filters.genre) params.set('genre', filters.genre);
   410→        if (filters.minPrice !== undefined && filters.minPrice > 0)
   411→          params.set('minPrice', String(filters.minPrice));
   412→        if (filters.maxPrice !== undefined && filters.maxPrice > 0)
   413→          params.set('maxPrice', String(filters.maxPrice));
   414→        if (filters.minRating !== undefined && filters.minRating > 0)
   415→          params.set('minRating', String(filters.minRating));
   416→        if (filters.sort) params.set('sort', filters.sort);
   417→        if (pageParams.search) params.set('search', pageParams.search);
   418→
   419→        const res = await fetch(`/api/books?${params.toString()}`);
   420→        if (res.ok && !cancelled) {
   421→          const data: BooksResponse = await res.json();
   422→          setBooks(data.books || []);
   423→          setTotal(data.total || 0);
   424→        }
   425→      } catch {
   426→        // silently fail
   427→      } finally {
   428→        if (!cancelled) setLoading(false);
   429→      }
   430→    };
   431→    doFetch();
   432→    return () => {
   433→      cancelled = true;
   434→    };
   435→  }, [filters, currentPage, pageParams.search]);
   436→
   437→  const totalPages = Math.max(1, Math.ceil(total / LIMIT));
   438→
   439→  const handleFilterChange = useCallback(
   440→    (newFilters: Record<string, unknown>) => {
   441→      setFilters(newFilters as Partial<FilterState>);
   442→    },
   443→    [setFilters]
   444→  );
   445→
   446→  const handleReset = useCallback(() => {
   447→    resetFilters();
   448→    setSearchInput('');
   449→    setCurrentPage(1);
   450→    setPage('catalog', {});
   451→  }, [resetFilters, setPage]);
   452→
   453→  const handleSearch = useCallback(
   454→    (e: React.FormEvent) => {
   455→      e.preventDefault();
   456→      setPage('catalog', { ...pageParams, search: searchInput.trim() });
   457→      setCurrentPage(1);
   458→    },
   459→    [searchInput, pageParams, setPage]
   460→  );
   461→
   462→  // Active filter badges
   463→  const activeFilters = useMemo(() => {
   464→    const items: { key: string; label: string; clear: () => void }[] = [];
   465→
   466→    if (filters.genre) {
   467→      const slugs = filters.genre.split(',').filter(Boolean);
   468→      slugs.forEach((slug) => {
   469→        const cat = categories.find((c) => c.slug === slug);
   470→        const name = cat
   471→          ? locale === 'en' && cat.nameEn
   472→            ? cat.nameEn
   473→            : cat.name
   474→          : slug;
   475→        items.push({
   476→          key: `genre-${slug}`,
   477→          label: name,
   478→          clear: () => {
   479→            const updated = slugs.filter((s) => s !== slug);
   480→            setFilters({ genre: updated.length > 0 ? updated.join(',') : undefined });
   481→          },
   482→        });
   483→      });
   484→    }
   485→
   486→    if (filters.minPrice !== undefined && filters.minPrice > 0) {
   487→      items.push({
   488→        key: 'minPrice',
   489→        label: `Min ${formatPrice(filters.minPrice, locale)}`,
   490→        clear: () => setFilters({ minPrice: undefined }),
   491→      });
   492→    }
   493→
   494→    if (filters.maxPrice !== undefined && filters.maxPrice > 0) {
   495→      items.push({
   496→        key: 'maxPrice',
   497→        label: `Max ${formatPrice(filters.maxPrice, locale)}`,
   498→        clear: () => setFilters({ maxPrice: undefined }),
   499→      });
   500→    }
   501→
   502→    if (filters.minRating !== undefined && filters.minRating > 0) {
   503→      items.push({
   504→        key: 'minRating',
   505→        label: `${filters.minRating}+ \u2605`,
   506→        clear: () => setFilters({ minRating: undefined }),
   507→      });
   508→    }
   509→
   510→    if (pageParams.search) {
   511→      items.push({
   512→        key: 'search',
   513→        label: `"${pageParams.search}"`,
   514→        clear: () => {
   515→          setSearchInput('');
   516→          setPage('catalog', { ...pageParams, search: '' });
   517→        },
   518→      });
   519→    }
   520→
   521→    return items;
   522→  }, [filters, categories, locale, pageParams, setFilters, setPage]);
   523→
   524→  // Breadcrumb: if single genre filtered, show the category name
   525→  const genreCategory = useMemo(() => {
   526→    if (!filters.genre) return null;
   527→    const slugs = filters.genre.split(',').filter(Boolean);
   528→    if (slugs.length === 1) {
   529→      return categories.find((c) => c.slug === slugs[0]) || null;
   530→    }
   531→    return null;
   532→  }, [filters.genre, categories]);
   533→
   534→  const hasActiveFilters = activeFilters.length > 0;
   535→
   536→  const filterSidebar = (
   537→    <FilterSidebar
   538→      categories={categories}
   539→      filters={filters}
   540→      onFilterChange={handleFilterChange}
   541→      onReset={handleReset}
   542→      locale={locale}
   543→    />
   544→  );
   545→
   546→  // Pagination range
   547→  const paginationRange = useMemo(() => {
   548→    const pages: (number | string)[] = [];
   549→    if (totalPages <= 7) {
   550→      for (let i = 1; i <= totalPages; i++) pages.push(i);
   551→    } else {
   552→      pages.push(1);
   553→      if (currentPage > 3) pages.push('...');
   554→      const start = Math.max(2, currentPage - 1);
