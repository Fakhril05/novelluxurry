    67→/** Generate simulated rating distribution from average */
    68→function getRatingDistribution(avgRating: number, totalReviews: number) {
    69→  // Create a bell-curve-like distribution based on the average rating
    70→  const base = totalReviews || 1;
    71→  const dist: number[] = [0, 0, 0, 0, 0]; // 5★, 4★, 3★, 2★, 1★
    72→  for (let i = 0; i < 5; i++) {
    73→    const starVal = 5 - i;
    74→    const diff = Math.abs(starVal - avgRating);
    75→    dist[i] = Math.max(5, Math.round(base * Math.exp(-diff * diff * 0.5)));
    76→  }
    77→  // Normalize so sum matches totalReviews
    78→  const sum = dist.reduce((a, b) => a + b, 0);
    79→  for (let i = 0; i < 5; i++) {
    80→    dist[i] = Math.round((dist[i] / sum) * base);
    81→  }
    82→  return dist;
    83→}
