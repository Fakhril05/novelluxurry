   349→   349→export default function CatalogPage() {
   350→   350→  const { locale, pageParams, setPage, filters, setFilters, resetFilters } =
   351→   351→    useAppStore();
   352→   352→
   353→   353→  const [books, setBooks] = useState<Book[]>([]);
   354→   354→  const [categories, setCategories] = useState<Category[]>([]);
   355→   355→  const [total, setTotal] = useState(0);
   356→   356→  const [currentPage, setCurrentPage] = useState(1);
   357→   357→  const [loading, setLoading] = useState(true);
   358→   358→  const [categoriesLoading, setCategoriesLoading] = useState(true);
   359→   359→  const [searchInput, setSearchInput] = useState('');
   360→   360→  const [sheetOpen, setSheetOpen] = useState(false);
   361→   361→
   362→   362→  // Sync genre from pageParams on mount
   363→   363→  useEffect(() => {
   364→   364→    if (pageParams.genre && !filters.genre) {
   365→   365→      setFilters({ genre: pageParams.genre });
   366→   366→    }
   367→   367→    if (pageParams.search) {
   368→   368→      setSearchInput(pageParams.search);
   369→   369→    }
   370→   370→  }, []);
   371→   371→
   372→   372→  // Fetch categories
   373→   373→  useEffect(() => {
   374→   374→    let cancelled = false;
   375→   375→    const fetchCategories = async () => {
   376→   376→      try {
   377→   377→        const res = await fetch('/api/categories');
   378→   378→        if (res.ok && !cancelled) {
   379→   379→          const data = await res.json();
   380→   380→          setCategories(Array.isArray(data) ? data : data.categories || []);
   381→   381→        }
   382→   382→      } catch {
   383→   383→        // silently fail
   384→   384→      } finally {
   385→   385→        if (!cancelled) setCategoriesLoading(false);
   386→   386→      }
   387→   387→    };
   388→   388→    fetchCategories();
   389→   389→    return () => {
   390→   390→      cancelled = true;
   391→   391→    };
   392→   392→  }, []);
   393→   393→
   394→   394→  // Reset to page 1 when filters change (except page itself)
   395→   395→  useEffect(() => {
   396→   396→    setCurrentPage(1);
   397→   397→  }, [filters.genre, filters.minPrice, filters.maxPrice, filters.minRating, filters.sort, pageParams.search]);
   398→   398→
   399→   399→  // Fetch books
   400→   400→  useEffect(() => {
   401→   401→    let cancelled = false;
   402→   402→    const doFetch = async () => {
   403→   403→      setLoading(true);
   404→   404→      try {
   405→   405→        const params = new URLSearchParams();
   406→   406→        params.set('limit', String(LIMIT));
   407→   407→        params.set('page', String(currentPage));
   408→   408→
   409→   409→        if (filters.genre) params.set('genre', filters.genre);
   410→   410→        if (filters.minPrice !== undefined && filters.minPrice > 0)
   411→   411→          params.set('minPrice', String(filters.minPrice));
   412→   412→        if (filters.maxPrice !== undefined && filters.maxPrice > 0)
   413→   413→          params.set('maxPrice', String(filters.maxPrice));
   414→   414→        if (filters.minRating !== undefined && filters.minRating > 0)
   415→   415→          params.set('minRating', String(filters.minRating));
   416→   416→        if (filters.sort) params.set('sort', filters.sort);
   417→   417→        if (pageParams.search) params.set('search', pageParams.search);
   418→   418→
   419→   419→        const res = await fetch(`/api/books?${params.toString()}`);
   420→   420→        if (res.ok && !cancelled) {
   421→   421→          const data: BooksResponse = await res.json();
   422→   422→          setBooks(data.books || []);
   423→   423→          setTotal(data.total || 0);
   424→   424→        }
   425→   425→      } catch {
   426→   426→        // silently fail
   427→   427→      } finally {
   428→   428→        if (!cancelled) setLoading(false);
   429→   429→      }
   430→   430→    };
   431→   431→    doFetch();
   432→   432→    return () => {
   433→   433→      cancelled = true;
   434→   434→    };
   435→   435→  }, [filters, currentPage, pageParams.search]);
   436→   436→
   437→   437→  const totalPages = Math.max(1, Math.ceil(total / LIMIT));
   438→   438→
   439→   439→  const handleFilterChange = useCallback(
   440→   440→    (newFilters: Record<string, unknown>) => {
   441→   441→      setFilters(newFilters as Partial<FilterState>);
   442→   442→    },
   443→   443→    [setFilters]
   444→   444→  );
   445→   445→
   446→   446→  const handleReset = useCallback(() => {
   447→   447→    resetFilters();
   448→   448→    setSearchInput('');
