   209→
   210→  const getCurrentTier = (points: number): TierInfo => {

   212→  };
   213→
   214→  const getNextTier = (points: number): TierInfo | null => {
   215→    const currentIdx = TIERS.findIndex((tier) => points >= tier.min && points < tier.max);
   216→    if (currentIdx === -1 || currentIdx === TIERS.length - 1) return null;
   217→    return TIERS[currentIdx + 1];

   220→  const getTierProgress = (points: number): number => {
   221→    const current = getCurrentTier(points);
   222→    const next = getNextTier(points);
   223→    if (!next) return 100;
   224→    const range = next.min - current.min;
   225→    const progress = points - current.min;
   226→    return Math.min(Math.round((progress / range) * 100), 100);
   227→  };

   230→    const date = new Date(dateStr);
