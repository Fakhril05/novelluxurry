    25→const SEARCH_HISTORY_KEY = 'noveluxe-search-history';
    26→const MAX_HISTORY = 8;
    27→
    28→function getSearchHistory(): string[] {
    29→  if (typeof window === 'undefined') return [];
    30→  try { return JSON.parse(localStorage.getItem(SEARCH_HISTORY_KEY) || '[]'); }
    31→  catch { return []; }
    32→}
    33→
    34→function addSearchHistory(query: string) {
    35→  if (!query.trim()) return;
    36→  try {
    37→    let history = getSearchHistory().filter((h) => h !== query.trim());
    38→    history.unshift(query.trim());
    39→    history = history.slice(0, MAX_HISTORY);
    40→    localStorage.setItem(SEARCH_HISTORY_KEY, JSON.stringify(history));
    41→  } catch { /* ignore */ }
    42→}
    43→
    44→function removeSearchHistoryItem(term: string) {
    45→  try {
    46→    const history = getSearchHistory().filter((h) => h !== term);
    47→    localStorage.setItem(SEARCH_HISTORY_KEY, JSON.stringify(history));
    48→  } catch { /* ignore */ }
    49→}
    50→
    51→function clearSearchHistory() {
    52→  try { localStorage.removeItem(SEARCH_HISTORY_KEY); } catch { /* ignore */ }
    53→}

    55→/** Highlight matching portions of text in gold */
    56→function HighlightText({ text, query }: { text: string; query: string }) {
    57→  if (!query.trim() || !text) return <>{text}</>;
    58→  const idx = text.toLowerCase().indexOf(query.toLowerCase());
    59→  if (idx === -1) return <>{text}</>;
    60→  const before = text.slice(0, idx);
    61→  const match = text.slice(idx, idx + query.length);
    62→  const after = text.slice(idx + query.length);
    63→  return (
    64→    <>
    65→      {before}
    66→      <span className="text-[#D4AF37] font-semibold">{match}</span>
    67→      {after}
    68→    </>
    69→  );
    70→}
    71→
    72→/** Small star rating display */
    73→function StarRating({ rating }: { rating: number }) {
    74→  if (!rating || rating <= 0) return null;
    75→  return (
    76→    <div className="flex items-center gap-0.5">
    77→      <Star className="h-3 w-3 fill-[#D4AF37] text-[#D4AF37]" />
    78→      <span className="text-[11px] font-medium text-muted-foreground">{rating.toFixed(1)}</span>
    79→    </div>
    80→  );
    81→}
    82→
    83→/** Category badge */
    84→function CategoryBadge({ category, locale }: { category?: { name: string; slug: string } | null; locale: Locale }) {
    85→  if (!category) return null;
    86→  return (
    87→    <span className="inline-flex items-center rounded-full bg-[#D4AF37]/10 px-2 py-0.5 text-[10px] font-medium text-[#D4AF37] border border-[#D4AF37]/20">
    88→      {locale === 'en' && category.slug ? category.slug.charAt(0).toUpperCase() + category.slug.slice(1) : category.name}
    89→    </span>
    90→  );
    91→}

'bg-[#D4AF37]/10 text-[#D4AF37]' : 'text-muted-foreground hover:bg-secondary hover:text-foreground'}`}>
   592→                  {t(item.labelKey, locale)}

   595→              <div className="my-2 border-t border-border" />
   596→              {isAuthenticated ? (
   597→                <>
   598→                  <button onClick={() => { setPage('dashboard'); setIsMobileMenuOpen(false); }} className="text-left px-4 py-2.5 rounded-lg text-sm font-medium text-muted-foreground hover:bg-secondary hover:text-foreground">{t('nav.profile', locale)}</button>
