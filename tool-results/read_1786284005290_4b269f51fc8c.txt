   190→function formatDate(dateStr: string, locale: string) {
   191→  const date = new Date(dateStr);
   192→  return date.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {

   397→function BooksTab({
   398→  locale,
   399→  books,
   400→  categories,
   401→  onRefresh,
   402→}: {
   403→  locale: string;
   404→  books: Book[];
   405→  categories: Category[];
   406→  onRefresh: () => void;
   407→}) {
   408→  const [dialogOpen, setDialogOpen] = useState(false);
   409→  const [editingBook, setEditingBook] = useState<Book | null>(null);
   410→  const [form, setForm] = useState<BookFormData>(EMPTY_BOOK_FORM);
   411→  const [saving, setSaving] = useState(false);
   412→  const [deletingId, setDeletingId] = useState<string | null>(null);
   413→
   414→  const openAddDialog = useCallback(() => {
   415→    setEditingBook(null);
   416→    setForm(EMPTY_BOOK_FORM);
   417→    setDialogOpen(true);
   418→  }, []);
   419→
   420→  const openEditDialog = useCallback((book: Book) => {
   421→    setEditingBook(book);
   422→    setForm({
   423→      title: book.title,
   424→      author: book.author,
   425→      authorBio: book.authorBio || '',
   426→      isbn: book.isbn || '',
   427→      synopsis: book.synopsis || '',
   428→      coverImage: book.coverImage,
   429→      price: String(book.price),
   430→      discountPrice: book.discountPrice ? String(book.discountPrice) : '',
   431→      stock: String(book.stock),
   432→      format: book.format,
   433→      pages: book.pages ? String(book.pages) : '',
   434→      publisher: book.publisher || '',
   435→      language: book.language,
   436→      publishedYear: book.publishedYear ? String(book.publishedYear) : '',
   437→      categoryId: book.categoryId || '',
   438→      isBestSeller: book.isBestSeller,
   439→      isNewArrival: book.isNewArrival,
   440→      isFeatured: book.isFeatured,
   441→    });
   442→    setDialogOpen(true);
   443→  }, []);
   444→
   445→  const handleSave = useCallback(async () => {
   446→    if (!form.title || !form.author || !form.price) {
   447→      toast.error(locale === 'id' ? 'Judul, penulis, dan harga wajib diisi' : 'Title, author, and price are required');
   448→      return;
   449→    }
   450→    setSaving(true);
   451→    try {
   452→      const payload = {
   453→        ...form,
   454→        price: Number(form.price),
   455→        discountPrice: form.discountPrice ? Number(form.discountPrice) : null,
   456→        stock: Number(form.stock),
   457→        pages: form.pages ? Number(form.pages) : null,
   458→        publishedYear: form.publishedYear ? Number(form.publishedYear) : null,
   459→      };
   460→
   461→      if (editingBook) {
   462→        const res = await fetch(`/api/books?id=${editingBook.id}`, {
   463→          method: 'PUT',
   464→          headers: { 'Content-Type': 'application/json' },
   465→          body: JSON.stringify(payload),
   466→        });
   467→        if (res.ok) {
   468→          toast.success(locale === 'id' ? 'Buku berhasil diperbarui' : 'Book updated successfully');
   469→          setDialogOpen(false);
   470→          onRefresh();
   471→        } else {
   472→          toast.error(locale === 'id' ? 'Gagal memperbarui buku' : 'Failed to update book');
   473→        }
   474→      } else {
   475→        const res = await fetch('/api/books', {
   476→          method: 'POST',
   477→          headers: { 'Content-Type': 'application/json' },
   478→          body: JSON.stringify(payload),
   479→        });
   480→        if (res.ok) {
   481→          toast.success(locale === 'id' ? 'Buku berhasil ditambahkan' : 'Book added successfully');
   482→          setDialogOpen(false);
   483→          onRefresh();
   484→        } else {
   485→          toast.error(locale === 'id' ? 'Gagal menambahkan buku' : 'Failed to add book');
   486→        }
   487→      }
   488→    } catch {
   489→      toast.error(locale === 'id' ? 'Terjadi kesalahan' : 'An error occurred');
   490→    } finally {
   491→      setSaving(false);
   492→    }
   493→  }, [form, editingBook, locale, onRefresh]);
   494→
   495→  const handleDelete = useCallback(async () => {
   496→    if (!deletingId) return;
   497→    try {
   498→      const res = await fetch(`/api/books?id=${deletingId}`, { method: 'DELETE' });
   499→      if (res.ok) {
   500→        toast.success(locale === 'id' ? 'Buku berhasil dihapus' : 'Book deleted successfully');
   501→        onRefresh();
   502→      } else {
   503→        toast.error(locale === 'id' ? 'Gagal menghapus buku' : 'Failed to delete book');
   504→      }
   505→    } catch {
   506→      toast.error(locale === 'id' ? 'Terjadi kesalahan' : 'An error occurred');
   507→    } finally {
   508→      setDeletingId(null);
   509→    }
   510→  }, [deletingId, locale, onRefresh]);
   511→
   512→  const getCategoryName = (catId: string | null) => {
   513→    if (!catId) return '-';
   514→    return categories.find((c) => c.id === catId)?.name || '-';
   515→  };
   516→
   517→  return (
   518→    <div className="space-y-4">
   519→      <div className="flex items-center justify-between">
   520→        <h2 className="font-heading text-xl font-bold">
   521→          {locale === 'id' ? 'Kelola Buku' : 'Manage Books'}
   522→        </h2>
   523→        <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   524→          <DialogTrigger asChild>
   525→            <Button onClick={openAddDialog} className="bg-[#D4AF37] hover:bg-[#D4AF37]/90 text-white">
   526→              <Plus className="h-4 w-4 mr-2" />
   527→              {locale === 'id' ? 'Tambah Buku' : 'Add Book'}
   528→            </Button>
   529→          </DialogTrigger>
   530→          <DialogContent className="max-h-[85vh] overflow-y-auto max-w-2xl">
   531→            <DialogHeader>
   532→              <DialogTitle className="font-heading">
   533→                {editingBook
   534→                  ? (locale === 'id' ? 'Edit Buku' : 'Edit Book')
   535→                  : (locale === 'id' ? 'Tambah Buku Baru' : 'Add New Book')}
   536→              </DialogTitle>
   537→            </DialogHeader>
   538→            <div className="grid gap-4 py-4">
   539→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   540→                <div className="space-y-2">
   541→                  <Label>{locale === 'id' ? 'Judul' : 'Title'} *</Label>
   542→                  <Input
   543→                    value={form.title}
   544→                    onChange={(e) => setForm((f) => ({ ...f, title: e.target.value }))}
   545→                    placeholder={locale === 'id' ? 'Judul buku' : 'Book title'}
   546→                  />
   547→                </div>
   548→                <div className="space-y-2">
   549→                  <Label>{locale === 'id' ? 'Penulis' : 'Author'} *</Label>
   550→                  <Input
   551→                    value={form.author}
   552→                    onChange={(e) => setForm((f) => ({ ...f, author: e.target.value }))}
   553→                    placeholder={locale === 'id' ? 'Nama penulis' : 'Author name'}
   554→                  />
   555→                </div>
   556→              </div>
   557→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   558→                <div className="space-y-2">
   559→                  <Label>ISBN</Label>
   560→                  <Input
   561→                    value={form.isbn}
   562→                    onChange={(e) => setForm((f) => ({ ...f, isbn: e.target.value }))}
   563→                    placeholder="978-xxx-xxx-xxx"
   564→                  />
   565→                </div>
   566→                <div className="space-y-2">
   567→                  <Label>{locale === 'id' ? 'Kategori' : 'Category'}</Label>
   568→                  <Select
   569→                    value={form.categoryId}
   570→                    onValueChange={(v) => setForm((f) => ({ ...f, categoryId: v }))}
   571→                  >
   572→                    <SelectTrigger className="w-full">
   573→                      <SelectValue placeholder={locale === 'id' ? 'Pilih kategori' : 'Select category'} />
   574→                    </SelectTrigger>
   575→                    <SelectContent>
   576→                      {categories.map((cat) => (

   877→function CategoriesTab({
   878→  locale,
   879→  categories,
   880→  onRefresh,
   881→}: {
   882→  locale: string;
   883→  categories: Category[];
   884→  onRefresh: () => void;
   885→}) {
   886→  const [dialogOpen, setDialogOpen] = useState(false);
   887→  const [editingCat, setEditingCat] = useState<Category | null>(null);
   888→  const [form, setForm] = useState<CategoryFormData>(EMPTY_CATEGORY_FORM);
   889→  const [saving, setSaving] = useState(false);
   890→  const [deletingId, setDeletingId] = useState<string | null>(null);
   891→
   892→  const openAddDialog = useCallback(() => {
   893→    setEditingCat(null);
   894→    setForm(EMPTY_CATEGORY_FORM);
   895→    setDialogOpen(true);
   896→  }, []);
   897→
   898→  const openEditDialog = useCallback((cat: Category) => {
   899→    setEditingCat(cat);
   900→    setForm({
   901→      name: cat.name,
   902→      slug: cat.slug,
   903→      description: cat.description || '',
   904→      image: cat.image || '',
   905→    });
   906→    setDialogOpen(true);
   907→  }, []);
   908→
   909→  const handleSave = useCallback(async () => {
   910→    if (!form.name || !form.slug) {
   911→      toast.error(locale === 'id' ? 'Nama dan slug wajib diisi' : 'Name and slug are required');
   912→      return;
   913→    }
   914→    setSaving(true);
   915→    try {
   916→      const payload = {
   917→        name: form.name,
   918→        nameEn: form.name,
   919→        slug: form.slug,
   920→        description: form.description || null,
   921→        image: form.image || null,
   922→      };
   923→
   924→      if (editingCat) {
   925→        const res = await fetch(`/api/categories?id=${editingCat.id}`, {
   926→          method: 'PUT',
   927→          headers: { 'Content-Type': 'application/json' },
   928→          body: JSON.stringify(payload),
   929→        });
   930→        if (res.ok) {
   931→          toast.success(locale === 'id' ? 'Kategori berhasil diperbarui' : 'Category updated');
   932→          setDialogOpen(false);
   933→          onRefresh();
   934→        } else {
   935→          toast.error(locale === 'id' ? 'Gagal memperbarui kategori' : 'Failed to update category');
   936→        }
   937→      } else {
   938→        const res = await fetch('/api/categories', {
   939→          method: 'POST',
   940→          headers: { 'Content-Type': 'application/json' },
   941→          body: JSON.stringify(payload),
   942→        });
   943→        if (res.ok) {
   944→          toast.success(locale === 'id' ? 'Kategori berhasil ditambahkan' : 'Category added');
   945→          setDialogOpen(false);
   946→          onRefresh();
   947→        } else {
   948→          toast.error(locale === 'id' ? 'Gagal menambahkan kategori' : 'Failed to add category');
   949→        }
   950→      }
   951→    } catch {
   952→      toast.error(locale === 'id' ? 'Terjadi kesalahan' : 'An error occurred');
   953→    } finally {
   954→      setSaving(false);
   955→    }
   956→  }, [form, editingCat, locale, onRefresh]);
   957→
   958→  const handleDelete = useCallback(async () => {

   964→        onRefresh();
   965→      } else {
   966→        toast.error(locale === 'id' ? 'Gagal menghapus kategori' : 'Failed to delete category');
   967→      }
   968→    } catch {
   969→      toast.error(locale === 'id' ? 'Terjadi kesalahan' : 'An error occurred');
   970→    } finally {
   971→      setDeletingId(null);
   972→    }
   973→  }, [deletingId, locale, onRefresh]);

   977→      <div className="flex items-center justify-between">
   978→        <h2 className="font-heading text-xl font-bold">
   979→          {locale === 'id' ? 'Kelola Kategori' : 'Manage Categories'}

   995→            </DialogHeader>

   997→              <div className="space-y-2">

   999→                <Input
