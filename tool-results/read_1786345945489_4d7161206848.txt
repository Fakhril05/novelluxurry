    98→function HorizontalBookRow({
    99→  books,
   100→  title,
   101→  subtitle,
   102→  viewAllLabel,
   103→  onViewAll,
   104→}: {
   105→  books: Book[];
   106→  title: string;
   107→  subtitle: string;
   108→  viewAllLabel: string;
   109→  onViewAll: () => void;
   110→}) {
   111→  const scrollRef = useRef<HTMLDivElement>(null);
   112→
   113→  const scroll = (dir: 'left' | 'right') => {
   114→    if (!scrollRef.current) return;
   115→    const amount = 340;
   116→    scrollRef.current.scrollBy({
   117→      left: dir === 'left' ? -amount : amount,
   118→      behavior: 'smooth',
   119→    });
   120→  };
   121→
   122→  if (books.length === 0) return null;
   123→
   124→  return (
   125→    <section className="py-16 sm:py-20">
   126→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   127→        <SectionHeader
   128→          title={title}
   129→          subtitle={subtitle}
   130→          action={{ label: viewAllLabel, onClick: onViewAll }}
   131→        />
   132→      </div>
   133→
   134→      <div className="relative group/row">
   135→        {/* Left arrow */}
   136→        <button
   137→          onClick={() => scroll('left')}
   138→          className="absolute left-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   139→          aria-label="Scroll left"
   140→        >
   141→          <ChevronLeft className="h-5 w-5" />
   142→        </button>
   143→        {/* Right arrow */}
   144→        <button
   145→          onClick={() => scroll('right')}
   146→          className="absolute right-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   147→          aria-label="Scroll right"
   148→        >
   149→          <ChevronRight className="h-5 w-5" />
   150→        </button>
   151→
   152→        <div
   153→          ref={scrollRef}
   154→          className="flex gap-5 overflow-x-auto scrollbar-hide px-4 sm:px-6 lg:px-8 pb-4 snap-x snap-mandatory"
   155→        >
   156→          {books.map((book, i) => (
   157→            <div key={book.id} className="w-[180px] sm:w-[200px] shrink-0 snap-start">
   158→              <BookCard book={book} index={i} />
   159→            </div>
   160→          ))}
   161→        </div>
   162→      </div>
   163→    </section>
   164→  );
   165→}
