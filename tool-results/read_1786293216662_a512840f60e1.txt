   272→// --- Helpers ---
   273→function formatDate(dateStr: string, locale: 'id' | 'en'): string {
   274→  const date = new Date(dateStr);
   275→  return date.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {
   276→    year: 'numeric',
   277→    month: 'long',
   278→    day: 'numeric',
   279→  });
   280→}
   281→
   282→function formatDateTime(dateStr: string, locale: 'id' | 'en'): string {
   283→  const date = new Date(dateStr);
   284→  return date.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {
   285→    year: 'numeric',
   286→    month: 'long',
   287→    day: 'numeric',
   288→    hour: '2-digit',
   289→    minute: '2-digit',
   290→  });
   291→}
