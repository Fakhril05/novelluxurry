    72→function estimateReadTime(content: string | null): number {
    73→  if (!content) return 1;
    74→  const text = content.replace(/<[^>]*>/g, '');
    75→  const wordCount = text.split(/\s+/).filter(Boolean).length;
    76→  return Math.max(1, Math.ceil(wordCount / 200));
    77→}

    79→function formatDate(dateStr: string, locale: Locale): string {
    80→  const date = new Date(dateStr);
    81→  return date.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {
    82→    year: 'numeric',
    83→    month: 'long',
    84→    day: 'numeric',
    85→  });
    86→}

    88→function getLocalizedField<T extends string | null>(
    89→  idValue: T,
    90→  enValue: T,
    91→  locale: Locale
    92→): T {
    93→  if (locale === 'en' && enValue) return enValue;
    94→  return idValue;
    95→}

   110→  'linear-gradient(135deg, #2d1b2e 0%, #44203a 50%, #6b2d5b 100%)',

   115→
   116→function getCategoryGradient(slug: string): string {
   117→  let hash = 0;
   118→  for (let i = 0; i < slug.length; i++) {
   119→    hash = slug.charCodeAt(i) + ((hash << 5) - hash);
   120→  }
   121→  return CATEGORY_GRADIENTS[Math.abs(hash) % CATEGORY_GRADIENTS.length];
   122→}

   501→export default function BlogPage() {
   502→  const { locale, pageParams, setPage } = useAppStore();
   503→  const [blogs, setBlogs] = useState<BlogPost[]>([]);
   504→  const [total, setTotal] = useState(0);
   505→  const [currentBlog, setCurrentBlog] = useState<BlogPost | null>(null);
   506→  const [loading, setLoading] = useState(true);
   507→  const [loadingMore, setLoadingMore] = useState(false);
   508→  const [detailLoading, setDetailLoading] = useState(false);
   509→  const [visibleCount, setVisibleCount] = useState(POSTS_PER_PAGE);
   510→  const [selectedCategory, setSelectedCategory] = useState('all');
   511→
   512→  const slug = pageParams.slug as string | undefined;
   513→
   514→  const fetchBlogs = useCallback(async () => {
   515→    try {
   516→      const res = await fetch('/api/blogs');
   517→      if (res.ok) {
   518→        const data = await res.json();
   519→        setBlogs(data.blogs || data);
   520→        setTotal(data.total ?? data.length ?? 0);
   521→      }
   522→    } catch {
   523→      // silently fail
   524→    } finally {
   525→      setLoading(false);
   526→    }
   527→  }, []);
   528→
   529→  useEffect(() => {
   530→    fetchBlogs();
   531→  }, [fetchBlogs]);
   532→
   533→  useEffect(() => {
   534→    if (slug) {
   535→      setDetailLoading(true);
   536→      async function fetchBlog() {
   537→        try {
   538→          const res = await fetch(`/api/blogs/${slug}`);
   539→          if (res.ok) {
   540→            const data = await res.json();
   541→            setCurrentBlog(data);
   542→          } else {
   543→            setCurrentBlog(null);
   544→          }
   545→        } catch {
   546→          setCurrentBlog(null);
   547→        } finally {
   548→          setDetailLoading(false);
   549→        }
   550→      }
   551→      fetchBlog();
   552→    } else {
   553→      setCurrentBlog(null);
   554→    }
   555→  }, [slug]);
   556→
   557→  // Collect all unique categories
   558→  const allCategories = useMemo(() => {
   559→    const cats = new Set<string>();
   560→    blogs.forEach((b) => {
   561→      const cat = getLocalizedField(b.category, b.categoryEn, locale);
   562→      if (cat) cats.add(cat);
   563→    });
   564→    return Array.from(cats);
   565→  }, [blogs, locale]);
   566→
   567→  // Filter blogs by category
   568→  const filteredBlogs = useMemo(() => {
   569→    if (selectedCategory === 'all') return blogs;
   570→    return blogs.filter((b) => {
   571→      const cat = getLocalizedField(b.category, b.categoryEn, locale);
   572→      return cat === selectedCategory;
   573→    });
   574→  }, [blogs, selectedCategory, locale]);
   575→
