   216→  const getCurrentTier = (points: number): TierInfo => {

   218→  };
   219→
   220→  const getNextTier = (points: number): TierInfo | null => {
   221→    const currentIdx = TIERS.findIndex((tier) => points >= tier.min && points < tier.max);
   222→    if (currentIdx === -1 || currentIdx === TIERS.length - 1) return null;
   223→    return TIERS[currentIdx + 1];

   226→  const getTierProgress = (points: number): number => {
   227→    const current = getCurrentTier(points);
   228→    const next = getNextTier(points);
   229→    if (!next) return 100;
   230→    const range = next.min - current.min;
   231→    const progress = points - current.min;
   232→    return Math.min(Math.round((progress / range) * 100), 100);
   233→  };

   236→    const date = new Date(dateStr);

   239→      month: 'short',
