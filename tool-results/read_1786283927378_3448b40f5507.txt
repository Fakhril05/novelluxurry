    93→    93→function HorizontalBookRow({
    94→    94→  books,
    95→    95→  title,

    97→    97→  viewAllLabel,
    98→    98→  onViewAll,
    99→    99→}: {
   100→   100→  books: Book[];
   101→   101→  title: string;
   102→   102→  subtitle: string;
   103→   103→  viewAllLabel: string;
   104→   104→  onViewAll: () => void;
   105→   105→}) {
   106→   106→  const scrollRef = useRef<HTMLDivElement>(null);
   107→   107→
   108→   108→  const scroll = (dir: 'left' | 'right') => {
   109→   109→    if (!scrollRef.current) return;
   110→   110→    const amount = 340;
   111→   111→    scrollRef.current.scrollBy({
   112→   112→      left: dir === 'left' ? -amount : amount,
   113→   113→      behavior: 'smooth',
   114→   114→    });
   115→   115→  };
   116→   116→
   117→   117→  if (books.length === 0) return null;
   118→   118→
   119→   119→  return (
   120→   120→    <section className="py-16 sm:py-20">
   121→   121→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   122→   122→        <SectionHeader
   123→   123→          title={title}
   124→   124→          subtitle={subtitle}
   125→   125→          action={{ label: viewAllLabel, onClick: onViewAll }}
   126→   126→        />
   127→   127→      </div>
   128→   128→
   129→   129→      <div className="relative group/row">
   130→   130→        {/* Left arrow */}
   131→   131→        <button
   132→   132→          onClick={() => scroll('left')}
   133→   133→          className="absolute left-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   134→   134→          aria-label="Scroll left"
   135→   135→        >
   136→   136→          <ChevronLeft className="h-5 w-5" />
   137→   137→        </button>
   138→   138→        {/* Right arrow */}
   139→   139→        <button
   140→   140→          onClick={() => scroll('right')}
   141→   141→          className="absolute right-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   142→   142→          aria-label="Scroll right"
   143→   143→        >
   144→   144→          <ChevronRight className="h-5 w-5" />
   145→   145→        </button>
   146→   146→
   147→   147→        <div
   148→   148→          ref={scrollRef}
   149→   149→          className="flex gap-5 overflow-x-auto scrollbar-hide px-4 sm:px-6 lg:px-8 pb-4 snap-x snap-mandatory"
   150→   150→        >
   151→   151→          {books.map((book, i) => (
   152→   152→            <div key={book.id} className="w-[180px] sm:w-[200px] shrink-0 snap-start">

   155→   155→          ))}
   156→   156→        </div>
   157→   157→      </div>
   158→   158→    </section>
   159→   159→  );
   160→   160→}
