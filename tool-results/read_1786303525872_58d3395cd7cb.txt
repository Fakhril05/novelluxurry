    65→function estimateReadTime(content: string | null): number {
    66→  if (!content) return 1;
    67→  const text = content.replace(/<[^>]*>/g, '');
    68→  const wordCount = text.split(/\s+/).filter(Boolean).length;
    69→  return Math.max(1, Math.ceil(wordCount / 200));
    70→}

    72→function formatDate(dateStr: string, locale: Locale): string {
    73→  const date = new Date(dateStr);
    74→  return date.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {
    75→    year: 'numeric',
    76→    month: 'long',
    77→    day: 'numeric',
    78→  });
    79→}
    80→
    81→function getLocalizedField<T extends string | null>(
    82→  idValue: T,
    83→  enValue: T,
    84→  locale: Locale
    85→): T {
    86→  if (locale === 'en' && enValue) return enValue;
    87→  return idValue;
    88→}

    90→function getAuthorInitials(author: string): string {
    91→  return author
    92→    .split(' ')
    93→    .filter(Boolean)
    94→    .map((w) => w[0])
    95→    .slice(0, 2)
    96→    .join('')
    97→    .toUpperCase();
    98→}

   379→  const [blogs, setBlogs] = useState<BlogPost[]>([]);
   380→  const [total, setTotal] = useState(0);
   381→  const [currentBlog, setCurrentBlog] = useState<BlogPost | null>(null);
   382→  const [loading, setLoading] = useState(true);
   383→  const [loadingMore, setLoadingMore] = useState(false);
   384→  const [detailLoading, setDetailLoading] = useState(false);
   385→  const [visibleCount, setVisibleCount] = useState(POSTS_PER_PAGE);
   386→  const [selectedCategory, setSelectedCategory] = useState('all');
   387→
   388→  const slug = pageParams.slug as string | undefined;
   389→
   390→  const fetchBlogs = useCallback(async () => {
   391→    try {
   392→      const res = await fetch('/api/blogs');
   393→      if (res.ok) {
   394→        const data = await res.json();
   395→        setBlogs(data.blogs || data);
   396→        setTotal(data.total ?? data.length ?? 0);
   397→      }
   398→    } catch {
   399→      // silently fail
   400→    } finally {
   401→      setLoading(false);
   402→    }
   403→  }, []);
   404→
   405→  useEffect(() => {
   406→    fetchBlogs();

   408→
