   557→                {/* Voucher */
   558→                }
   559→                <div className="mb-4">
   560→                  <div className="flex gap-2">
   561→                    <div className="relative flex-1">
   562→                      <Tag className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   563→                      <Input

   567→                          if (voucherApplied) {
   568→                            setVoucherApplied(false);
   569→                            setVoucherDiscount(0);
   570→                            setVoucherError('');
   571→                          }
   572→                        }}
   573→                        placeholder={t('cart.voucher', lang)}
   574→                        className="pl-8 h-9 text-sm"
   575→                        disabled={voucherApplied}
   576→                        onKeyDown={(e) => e.key === 'Enter' && handleApplyVoucher()}
   577→                      />
   578→                    </div>
   579→                    <Button
   580→                      variant="outline"
   581→                      size="sm"
   582→                      onClick={handleApplyVoucher}
   583→                      disabled={voucherLoading || voucherApplied || !voucherCode.trim()}
   584→                      className="h-9 border-[#D4AF37] text-[#D4AF37] hover:bg-[#D4AF37]/10 shrink-0 px-3"
   585→                    >
   586→                      {voucherLoading
   587→                        ? t('general.loading', lang)
   588→                        : voucherApplied
   589→                          ? '✓'
   590→                          : t('cart.applyVoucher', lang)}
   591→                    </Button>
   592→                  </div>
   593→                  {voucherError && (
   594→                    <p className="mt-1 text-xs text-red-500">{voucherError}</p>
   595→                  )}
   596→                  {voucherApplied && voucherDiscount > 0 && (
   597→                    <motion.p
   598→                      initial={{ opacity: 0, y: -4 }}
   599→                      animate={{ opacity: 1, y: 0 }}
   600→                      className="mt-1 text-xs text-green-600 dark:text-green-400"
   601→                    >
   602→                      {lang === 'id' ? 'Voucher aktif' : 'Voucher active'}: -{formatPrice(voucherDiscount, lang)}
   603→                    </motion.p>
   604→                  )}
   605→                </div>
   606→
   607→                <Separator className="my-4" />
   608→
   609→                {/* Price Breakdown */
   610→                }
   611→                <div className="flex flex-col gap-2.5">
   612→                  <div className="flex items-center justify-between text-sm">
   613→                    <span className="text-muted-foreground">{t('cart.subtotal', lang)}</span>
   614→                    <span className="font-medium">{formatPrice(subtotal, lang)}</span>
   615→                  </div>
   616→                  {savings > 0 && (
   617→                    <div className="flex items-center justify-between text-sm">
   618→                      <span className="text-green-600 dark:text-green-400">{t('cart.discount', lang)}</span>
   619→                      <span className="text-green-600 dark:text-green-400 font-medium">
   620→                        -{formatPrice(savings, lang)}
   621→                      </span>
   622→                    </div>
   623→                  )}
   624→                  {voucherApplied && voucherDiscount > 0 && (
   625→                    <div className="flex items-center justify-between text-sm">
   626→                      <span className="text-green-600 dark:text-green-400">Voucher</span>
   627→                      <span className="text-green-600 dark:text-green-400 font-medium">
   628→                        -{formatPrice(voucherDiscount, lang)}
   629→                      </span>
   630→                    </div>
   631→                  )}
   632→                  <div className="flex items-center justify-between text-sm">
   633→                    <span className="text-muted-foreground">{t('cart.shipping', lang)}</span>
   634→                    <div className="text-right">
   635→                      <span className="block font-medium">{formatPrice(shippingCost, lang)}</span>
   636→                      {(() => {
   637→                        const selectedExp = EXPEDITIONS.find(e => e.id === expedition);
   638→                        return selectedExp ? (
   639→                          <span className="text-[10px] text-muted-foreground">
   640→                            {t('checkout.shippingNote', lang)}: {lang === 'id' ? selectedExp.eta : selectedExp.etaEn}
   641→                          </span>
   642→                        ) : null;
   643→                      })()}
   644→                    </div>
   645→                  </div>
   646→
   647→                  <Separator />
   648→
   649→                  <div className="flex items-center justify-between">
   650→                    <span className="font-semibold">{t('cart.total', lang)}</span>
   651→                    <span className="text-xl font-bold text-[#D4AF37]">{formatPrice(total, lang)}</span>
   652→                  </div>
   653→                </div>
   654→
   655→                {/* Points earned */
   656→                }
   657→                {total >= 10000 && (
   658→                  <div className="mt-2 flex items-center justify-between rounded-lg bg-[#D4AF37]/5 px-3 py-2 text-sm">
   659→                    <span className="text-muted-foreground">{t('checkout.pointsEarned', lang)}</span>
   660→                    <span className="font-bold text-[#D4AF37]">
   661→                      +{Math.floor(total / 10000)} {t('points.label', lang)}
   662→                    </span>
   663→                  </div>
   664→                )}
   665→
   666→                {/* Place Order Button */
   667→                }
   668→                <Button
   669→                  onClick={handleSubmit}
   670→                  disabled={submitting}
   671→                  className="mt-5 h-12 w-full bg-[#D4AF37] text-black font-semibold text-sm hover:bg-[#C4A030] disabled:opacity-60"
   672→                >
   673→                  {submitting ? (
   674→                    <span className="flex items-center gap-2">
   675→                      <span className="inline-block h-4 w-4 animate-spin rounded-full border-2 border-black/30 border-t-black" />
   676→                      {t('general.loading', lang)}
   677→                    </span>
   678→                  ) : (
   679→                    t('checkout.placeOrder', lang)
   680→                  )}
   681→                </Button>
   682→              </CardContent>
   683→            </Card>
   684→          </motion.div>
   685→        </div>
   686→      </div>
   687→    </div>
   688→  );
