   272→// --- Helpers ---
   273→function formatDate(dateStr: string, locale: 'id' | 'en'): string {
   274→  const date = new Date(dateStr);
   275→  return date.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {
   276→    year: 'numeric',
   277→    month: 'long',
   278→    day: 'numeric',
   279→  });
   280→}
   281→
   282→function formatDateTime(dateStr: string, locale: 'id' | 'en'): string {
   283→  const date = new Date(dateStr);
   284→  return date.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {
   285→    year: 'numeric',
   286→    month: 'long',
   287→    day: 'numeric',
   288→    hour: '2-digit',
   289→    minute: '2-digit',
   290→  });
   291→}
   292→
   293→function getPaymentLabel(method: string | null, locale: 'id' | 'en'): string {
   294→  if (!method) return '-';
   295→  const labels: Record<string, { id: string; en: string }> = {
   296→    'bank-transfer': { id: 'Transfer Bank', en: 'Bank Transfer' },
   297→    qris: { id: 'QRIS', en: 'QRIS' },
   298→    'e-wallet': { id: 'E-Wallet', en: 'E-Wallet' },
   299→    'virtual-account': { id: 'Virtual Account', en: 'Virtual Account' },
   300→    cod: { id: 'COD (Bayar di Tempat)', en: 'COD (Cash on Delivery)' },
   301→  };
   302→  return labels[method]?.[locale] || method;
   303→}
   304→
   305→function getEstimatedDate(orderDate: string, daysOffset: number, locale: 'id' | 'en'): string {
   306→  const d = new Date(orderDate);
   307→  d.setDate(d.getDate() + daysOffset);
   308→  return d.toLocaleDateString(locale === 'id' ? 'id-ID' : 'en-US', {
   309→    weekday: 'short', month: 'short', day: 'numeric',
   310→  });
   311→}
