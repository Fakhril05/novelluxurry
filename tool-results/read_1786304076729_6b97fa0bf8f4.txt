    95→function HorizontalBookRow({
    96→  books,
    97→  title,
    98→  subtitle,
    99→  viewAllLabel,
   100→  onViewAll,
   101→}: {
   102→  books: Book[];
   103→  title: string;
   104→  subtitle: string;
   105→  viewAllLabel: string;
   106→  onViewAll: () => void;
   107→}) {
   108→  const scrollRef = useRef<HTMLDivElement>(null);
   109→
   110→  const scroll = (dir: 'left' | 'right') => {
   111→    if (!scrollRef.current) return;
   112→    const amount = 340;
   113→    scrollRef.current.scrollBy({
   114→      left: dir === 'left' ? -amount : amount,
   115→      behavior: 'smooth',
   116→    });
   117→  };
   118→
   119→  if (books.length === 0) return null;
   120→
   121→  return (
   122→    <section className="py-16 sm:py-20">
   123→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   124→        <SectionHeader
   125→          title={title}
   126→          subtitle={subtitle}
   127→          action={{ label: viewAllLabel, onClick: onViewAll }}
   128→        />
   129→      </div>
   130→
   131→      <div className="relative group/row">
   132→        {/* Left arrow */}
   133→        <button
   134→          onClick={() => scroll('left')}
   135→          className="absolute left-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   136→          aria-label="Scroll left"
   137→        >
   138→          <ChevronLeft className="h-5 w-5" />
   139→        </button>
   140→        {/* Right arrow */}
   141→        <button
   142→          onClick={() => scroll('right')}
   143→          className="absolute right-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   144→          aria-label="Scroll right"
   145→        >
   146→          <ChevronRight className="h-5 w-5" />
   147→        </button>
   148→
   149→        <div
   150→          ref={scrollRef}
   151→          className="flex gap-5 overflow-x-auto scrollbar-hide px-4 sm:px-6 lg:px-8 pb-4 snap-x snap-mandatory"
   152→        >
   153→          {books.map((book, i) => (
   154→            <div key={book.id} className="w-[180px] sm:w-[200px] shrink-0 snap-start">
   155→              <BookCard book={book} index={i} />

   161→  );
