    95→    95→function HorizontalBookRow({
    96→    96→  books,
    97→    97→  title,
    98→    98→  subtitle,
    99→    99→  viewAllLabel,
   100→   100→  onViewAll,
   101→   101→}: {
   102→   102→  books: Book[];
   103→   103→  title: string;
   104→   104→  subtitle: string;
   105→   105→  viewAllLabel: string;
   106→   106→  onViewAll: () => void;
   107→   107→}) {
   108→   108→  const scrollRef = useRef<HTMLDivElement>(null);
   109→   109→
   110→   110→  const scroll = (dir: 'left' | 'right') => {
   111→   111→    if (!scrollRef.current) return;
   112→   112→    const amount = 340;
   113→   113→    scrollRef.current.scrollBy({
   114→   114→      left: dir === 'left' ? -amount : amount,
   115→   115→      behavior: 'smooth',
   116→   116→    });
   117→   117→  };
   118→   118→
   119→   119→  if (books.length === 0) return null;
   120→   120→
   121→   121→  return (
   122→   122→    <section className="py-16 sm:py-20">
   123→   123→      <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   124→   124→        <SectionHeader
   125→   125→          title={title}
   126→   126→          subtitle={subtitle}
   127→   127→          action={{ label: viewAllLabel, onClick: onViewAll }}
   128→   128→        />
   129→   129→      </div>
   130→   130→
   131→   131→      <div className="relative group/row">
   132→   132→        {/* Left arrow */}
   133→   133→        <button
   134→   134→          onClick={() => scroll('left')}
   135→   135→          className="absolute left-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   136→   136→          aria-label="Scroll left"
   137→   137→        >
   138→   138→          <ChevronLeft className="h-5 w-5" />
   139→   139→        </button>
   140→   140→        {/* Right arrow */}
   141→   141→        <button
   142→   142→          onClick={() => scroll('right')}
   143→   143→          className="absolute right-2 top-1/2 -translate-y-1/2 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/90 border border-border shadow-lg opacity-0 group-hover/row:opacity-100 transition-opacity hover:bg-[#D4AF37] hover:text-white hover:border-[#D4AF37] hidden sm:flex"
   144→   144→          aria-label="Scroll right"
   145→   145→        >
   146→   146→          <ChevronRight className="h-5 w-5" />
   147→   147→        </button>
   148→   148→
   149→   149→        <div
   150→   150→          ref={scrollRef}
   151→   151→          className="flex gap-5 overflow-x-auto scrollbar-hide px-4 sm:px-6 lg:px-8 pb-4 snap-x snap-mandatory"
   152→   152→        >
   153→   153→          {books.map((book, i) => (
   154→   154→            <div key={book.id} className="w-[180px] sm:w-[200px] shrink-0 snap-start">
   155→   155→              <BookCard book={book} index={i} />
   156→   156→            </div>
   157→   157→          ))}
   158→   158→        </div>
   159→   159→      </div>
   160→   160→    </section>
   161→   161→  );
   162→   162→}
